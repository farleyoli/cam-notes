This part establishes the language used throughout the paper.  All sets in

\concept{Proofs and statements}
A \emph{proof} is a succession of true statements, without logical gaps, which
establishes a conclusion from stated assumptions.  An assumption accepted as
a starting point is called an \emph{axiom}.  A \emph{statement} is a sentence
which is either true, denoted by $\mathrm T$, or false, denoted by
$\mathrm F$.

\concept{Logical symbols}
For statements $P$ and $Q$, the symbols $P\wedge Q$, $P\vee Q$,
$P\Rightarrow Q$, $P\Longleftrightarrow Q$, and $\neg P$ mean respectively
``$P$ and $Q$'', ``$P$ or $Q$'', ``$P$ implies $Q$'', ``$P$ if and only if
$Q$'', and ``not $P$''.  Here ``or'' is inclusive.  The implication
$P\Rightarrow Q$ is false only when $P$ is true and $Q$ is false.
A \emph{truth table} lists the truth value of a compound statement for each
possible assignment of $\mathrm T$ or $\mathrm F$ to its component
statements.

\concept{Sets and quantifiers}
A \emph{set} is a collection of objects, called its \emph{elements}, in
which order is irrelevant and repetitions are ignored.  We
write $x\in X$ when $x$ is an element of the set $X$, and $x\notin X$
otherwise.  If $P(x)$ is a statement for each $x\in X$, then
$(\forall x\in X)P(x)$ means that $P(x)$ is true for every $x\in X$, while
$(\exists x\in X)P(x)$ means that it is true for at least one $x\in X$.
Negating these quantifiers gives
\[
  \neg(\forall x\in X)P(x)\Longleftrightarrow
    (\exists x\in X)\neg P(x),\qquad
  \neg(\exists x\in X)P(x)\Longleftrightarrow
    (\forall x\in X)\neg P(x).
\]

For the final subquestion below, $\N=\{1,2,3,\ldots\}$ denotes the set of
positive integers.  If $d,n\in\N$, then $d$ is a \emph{positive divisor}
of $n$ if $n=dk$ for some $k\in\N$.  A number $p\in\N$ is \emph{prime}
if $p>1$ and its only positive divisors are $1$ and $p$.
The assertion that there are \emph{infinitely many primes} means that, for
every $n\in\N$, there is a prime $p\in\N$ with $p>n$.

\examquestion{\label{q:logic}
Let $A,B,C$ be sets, let $P,Q$ be statements, and suppose that $R(a,b,c)$
is a statement for every $a\in A$, $b\in B$, and $c\in C$.
\begin{subquestions}
  \item \nameditem{ITT}{Implication Truth Table}
  Write the truth table for $P\Rightarrow Q$.
  \item \nameditem{TTLE}{Truth Table Logical Equivalences}
  Prove, using truth tables, that
  \[
    (P\Rightarrow Q)\ \Longleftrightarrow\
    (\neg Q\Rightarrow\neg P)
    \quad\text{and}\quad
    \neg(P\wedge Q)\ \Longleftrightarrow\ (\neg P\vee\neg Q).
  \]
  \item \nameditem{NQN}{Nested Quantifier Negation}
  Negate the statement
  \[
    (\forall a\in A)(\exists b\in B)(\forall c\in C)\ R(a,b,c)
  \]
  without placing a negation sign in front of a quantifier.
  \item \nameditem{IMPQF}{Infinitely Many Primes in Quantifier Form}
  Express in symbols, and then negate, the assertion: for every
  $n\in\N$, there exists a prime $p$ with $p>n$.
\end{subquestions}}

\concept{Integers, parity, and proof methods}
The set of integers is
\[
  \Z=\{\ldots,-2,-1,0,1,2,\ldots\},
\]
and $\R$ denotes the set of real numbers.  An integer $n\in\Z$ is
\emph{even} if $n=2k$ for some
$k\in\Z$, and is \emph{odd} if $n=2k+1$ for some $k\in\Z$.
The \emph{contrapositive} of $P\Rightarrow Q$ is
$\neg Q\Rightarrow\neg P$.  A proof by contraposition proves an
implication by proving its contrapositive.  A proof by contradiction assumes
that the desired conclusion is false and derives a false statement.  A
\emph{counterexample} is an example for which a proposed universal statement
is false.

\examquestion{\label{q:proofs}
Let $n\in\Z$.  Consider the assertion: if $n^2$ is even, then $n$ is even.
\begin{subquestions}
  \item \nameditem{FPD}{Flawed Proof Direction}
  Explain why the calculation
  $n=2k\Rightarrow n^2=4k^2$, where $k\in\Z$, does not prove the assertion.
  \item \nameditem{ESP}{Even Squares and Parity}
  Prove the assertion by contraposition.  Take care to write an odd
  square in the form $2m+1$ with $m\in\Z$.
  \item \nameditem{LPRF}{Least Positive Real Fallacy}
  An argument begins by letting $r$ be the least positive real number
  and purports to prove that every positive real is at least $1$.  Give a
  counterexample to the conclusion and identify the unjustified step.
\end{subquestions}}

\concept{Set notation and operations}
Two sets are \emph{equal}, written $A=B$, when
\[
  (\forall x)(x\in A\Longleftrightarrow x\in B).
\]
For sets $A$ and $B$, write $A\subseteq B$ and say that $A$ is a
\emph{subset} of $B$ when
\[
  (\forall x)(x\in A\Rightarrow x\in B).
\]
The symbol $\varnothing$ denotes the \emph{empty set}, which has no
elements.  If $X$ is a set and $P(x)$ is a statement for each $x\in X$, then
$\{x\in X:P(x)\}$ denotes the subset of elements for which $P(x)$ is true.

If $X$ is a set and $A,B\subseteq X$, define
\[
  A\cap B=\{x\in X:x\in A\wedge x\in B\},\qquad
  A\cup B=\{x\in X:x\in A\vee x\in B\},
\]
\[
  A\setminus B=\{x\in A:x\notin B\},\qquad
  A\triangle B=(A\setminus B)\cup(B\setminus A).
\]
These are respectively the \emph{intersection}, \emph{union}, \emph{set
difference}, and \emph{symmetric difference}.  If $X$ and $I$ are sets, a
family $(A_i)_{i\in I}$ of subsets of $X$ is a choice of a subset
$A_i\subseteq X$ for each element $i$ of the \emph{index set} $I$.  Its
union and intersection are
\[
  \bigcup_{i\in I}A_i=\{x\in X:(\exists i\in I)\ x\in A_i\},\qquad
  \bigcap_{i\in I}A_i=\{x\in X:(\forall i\in I)\ x\in A_i\}.
\]
For objects $a$ and $b$, the notation $\{a\}$ denotes the set whose only
element is $a$, and $\{a,b\}$ denotes the set whose elements are $a$ and
$b$; if $a=b$, this latter set has only one element.

\examquestion{\label{q:sets}
Let $X$ and $I$ be sets, let $A,B,C\subseteq X$, and, for every $i\in I$,
let $A_i\subseteq X$.
Complements are relative to $X$, and the empty-family conventions are
$\bigcup_{i\in\varnothing}A_i=\varnothing$ and
$\bigcap_{i\in\varnothing}A_i=X$.
\begin{subquestions}
  \item \nameditem{SEC}{Set Equality Criterion}
  Prove that $A=B$ if and only if $A\subseteq B$ and $B\subseteq A$.
  \item \nameditem{DMLIF}{De Morgan Laws for Indexed Families}
  Prove the two De Morgan identities
  \[
    X\setminus\bigcup_{i\in I}A_i
      =\bigcap_{i\in I}(X\setminus A_i),
    \qquad
    X\setminus\bigcap_{i\in I}A_i
      =\bigcup_{i\in I}(X\setminus A_i).
  \]
  \item \nameditem{DIU}{Distribution over an Indexed Union}
  Prove directly from membership that
  \[
    A\cap\biggl(\bigcup_{i\in I}A_i\biggr)
      =\bigcup_{i\in I}(A\cap A_i).
  \]
  \item \nameditem{ASD}{Associativity of Symmetric Difference}
  Prove
  $(A\triangle B)\triangle C=A\triangle(B\triangle C)$.
  \item \nameditem{RUSP}{Russell's Universal Set Paradox}
  Suppose, contrary to the usual axioms of set theory, that there is a
  set $U$ containing every set and that every property defines a subset of
  $U$.  Put $R=\{x\in U:x\notin x\}$.  Show that
  $R\in R\Longleftrightarrow R\notin R$, and explain what this paradox rules
  out.
\end{subquestions}}

\examquestion{\label{q:ordered-pairs}
Let $X$ be a set and let $a,b,c,d\in X$.  Define the ordered pair by
\[
  (a,b)=\bigl\{\{a\},\{a,b\}\bigr\}.
\]
\begin{subquestions}
  \item \nameditem{OPEFD}{Ordered Pair Equality in the Forward Direction}
  Show that $(a,b)=(c,d)$ whenever $a=c$ and $b=d$.
  \item \nameditem{OPECD}{Ordered Pair Equality in the Converse Direction}
  Prove the converse.
\end{subquestions}
\indication{Separate the case $a=b$ from the case $a\ne b$.  In the second
case the displayed set has two distinct members, one of which is a
  set with exactly one element.}}

\concept{Cartesian products and functions}
For sets $A$ and $B$, their \emph{Cartesian product} is
\[
  A\times B=\{(a,b):a\in A\wedge b\in B\}.
\]
A \emph{function}, or \emph{map}, $f\colon A\to B$ is a subset
$f\subseteq A\times B$ such that, for every $a\in A$, there is exactly one
$b\in B$ with $(a,b)\in f$.  This unique $b$ is denoted by $f(a)$, so that
$f$ assigns to every $a\in A$ exactly one element $f(a)\in B$.  The sets
$A$ and $B$ are called the \emph{domain} and \emph{codomain} of $f$, and
the subset $f$ is called its \emph{graph}.

\examquestion{\label{q:products}
Let $A,B,C$ be sets.
\begin{subquestions}
  \item \nameditem{CPE}{Cartesian Product Emptiness}
  Prove that $A\times B=\varnothing$ if and only if $A=\varnothing$ or
  $B=\varnothing$.
  \item \nameditem{CPFC}{Cartesian Product Factor Cancellation}
  Assume $A\ne\varnothing$.  Prove that
  $A\times B=A\times C$ implies $B=C$.
  \item \nameditem{EFC}{Empty-Factor Counterexample}
  Show by an example that the hypothesis $A\ne\varnothing$ cannot be
  omitted.
\end{subquestions}}

\concept{Images and pre-images}
Let $f\colon A\to B$ be a function.  For $U\subseteq A$, the
\emph{image} of $U$ is
$f(U)=\{f(u):u\in U\}$.  For $V\subseteq B$, the \emph{pre-image} of
$V$ is
$f^{-1}(V)=\{a\in A:f(a)\in V\}$.  This pre-image notation does not
assert that an inverse function exists.

\concept{Injective functions}
A function $f\colon A\to B$ is \emph{injective} if, for all
$a,a'\in A$, the equality $f(a)=f(a')$ implies $a=a'$.

\examquestion{\label{q:image-preimage}
Let $A,B,I,J$ be sets, let $f\colon A\to B$, let $(U_i)_{i\in I}$ be a
family of subsets of $A$, and let $(V_j)_{j\in J}$ be a family of subsets
of $B$.  Also let $U_1,U_2\subseteq A$.
\begin{subquestions}
  \item \nameditem{IPU}{Images and Pre-images of Unions}
  Prove that
  \[
    f\left(\bigcup_{i\in I}U_i\right)=\bigcup_{i\in I}f(U_i),
    \qquad
    f^{-1}\left(\bigcup_{j\in J}V_j\right)
      =\bigcup_{j\in J}f^{-1}(V_j).
  \]
  \item \nameditem{PCI}{Pre-images of Complements and Intersections}
  Prove that, for every $V\subseteq B$,
  \[
    f^{-1}(B\setminus V)=A\setminus f^{-1}(V),
  \]
  and that
  \[
    f^{-1}\left(\bigcap_{j\in J}V_j\right)
      =\bigcap_{j\in J}f^{-1}(V_j).
  \]
  \item \nameditem{III}{Image of an Intersection Inclusion}
  Prove that
  $f(U_1\cap U_2)\subseteq f(U_1)\cap f(U_2)$, and give an example where the
  inclusion is strict.
  \item \nameditem{IECI}{Intersection Equality Characterizes Injectivity}
  Prove that equality holds in the preceding inclusion for every pair
  $U_1,U_2\subseteq A$ if and only if $f$ is injective.
\end{subquestions}}

\concept{Surjective and bijective functions}
A function $f\colon A\to B$ is \emph{surjective} if, for every $b\in B$,
there is an $a\in A$ with $f(a)=b$.  It is \emph{bijective} if it is both
injective and surjective.

\concept{Composition and identity functions}
If $f\colon A\to B$ and $g\colon B\to C$ are functions, their
\emph{composition} is the function $g\circ f\colon A\to C$ defined by
$(g\circ f)(a)=g(f(a))$ for every $a\in A$.  The composition is read from
right to left.  The \emph{identity function} on a set $A$ is
$\operatorname{id}_A\colon A\to A$, defined by
$\operatorname{id}_A(a)=a$ for every $a\in A$.  Two functions with the same
domain and codomain are equal if and only if they have the same value at
every element of their domain.

\examquestion{\label{q:map-criteria}
Let $A,B,C,D$ be sets and let $f\colon A\to B$, $g\colon B\to C$, and
$h\colon C\to D$ be functions.
\begin{subquestions}
  \item \nameditem{PICI}{Pre-image--Image Criterion for Injectivity}
  Prove that $U\subseteq f^{-1}(f(U))$ for every $U\subseteq A$, and
  that equality holds for every $U$ if and only if $f$ is injective.
  \item \nameditem{IPCS}{Image--Pre-image Criterion for Surjectivity}
  Prove that $f(f^{-1}(V))\subseteq V$ for every $V\subseteq B$, and
  that equality holds for every $V$ if and only if $f$ is surjective.
  \item \nameditem{CPIS}{Composition Preserves Injectivity and Surjectivity}
  Show that composition of injective maps is injective and composition
  of surjective maps is surjective.
  \item \nameditem{CAIL}{Composition Associativity and Identity Laws}
  Prove $h\circ(g\circ f)=(h\circ g)\circ f$, and prove the identity
  laws
  \[
    f\circ\operatorname{id}_A=f
    \qquad\text{and}\qquad
    \operatorname{id}_B\circ f=f.
  \]
\end{subquestions}}

\concept{Inverse functions and choice}
Let $f\colon A\to B$ be a function.  A \emph{left inverse} of $f$ is a
function $g\colon B\to A$ such that
$g\circ f=\operatorname{id}_A$; a \emph{right inverse} satisfies
$f\circ g=\operatorname{id}_B$.  A \emph{two-sided inverse} satisfies
both identities.

The \emph{axiom of choice} states that, if $I$ is a set and
$(E_i)_{i\in I}$ is a family of non-empty sets, then there is a function
\[
  c\colon I\longrightarrow\bigcup_{i\in I}E_i
\]
such that $c(i)\in E_i$ for every $i\in I$.  Such a function is called a
\emph{choice function} for the family.  Here
$\bigcup_{i\in I}E_i$ denotes the set of all objects $x$ for which
$x\in E_i$ for at least one $i\in I$.

\examquestion{\label{q:inverses}
Let $A$ and $B$ be sets, assume $A\ne\varnothing$, and let
$f\colon A\to B$ be a function.
\begin{subquestions}
  \item \nameditem{IIIS}{Inverses Imply Injectivity and Surjectivity}
  Show that a left inverse of $f$ forces $f$ to be injective, and that
  a right inverse forces $f$ to be surjective.
  \item \nameditem{ILI}{Injectivity gives a Left Inverse}
  If $f$ is injective, construct a left inverse.
  \item \nameditem{SRI}{Surjectivity gives a Right Inverse}
  Assuming the axiom of choice, prove that, if $f$ is surjective,
  then $f$ has a right inverse.
  \item \nameditem{BUI}{Bijection has a Unique Inverse}
  Prove that $f$ has a two-sided inverse if and only if it is
  bijective, and prove that this inverse is unique.
  \item \nameditem{EDLIO}{Empty Domain and Left Inverse Obstruction}
  Explain why the non-emptiness hypothesis is relevant to the
  converse concerning left inverses.
  \item \nameditem{ACRI}{Axiom of Choice and Right Inverses}
  Conversely, suppose that every surjective function has a right
  inverse.  Let $I$ be a set and let $(E_i)_{i\in I}$ be a family of
  non-empty sets.  Put
  \[
    D=\left\{(i,x)\in I\times\bigcup_{j\in I}E_j:x\in E_i\right\}
  \]
  and define $\pi\colon D\to I$ by $\pi(i,x)=i$ for every
  $(i,x)\in D$.  Prove that $\pi$ is surjective, and use a right inverse
  of $\pi$ to construct a choice function for $(E_i)_{i\in I}$.  Thus
  prove that the assertion that every surjection has a right inverse is
  equivalent to the axiom of choice.
\end{subquestions}}
\indication{For the construction of a left inverse, choose an element
$a_0\in A$ and send every member of $B\setminus f(A)$ to $a_0$.  For a
right inverse, choose one element of each non-empty pre-image
$f^{-1}(\{b\})$, where $b\in B$.  In the last part, if a right inverse
$s\colon I\to D$ has $s(i)=(i,x_i)$, set $c(i)=x_i$.}

\concept{Relations, equivalence classes, and partitions}
For a set $A$, a \emph{relation} on $A$ is a subset of $A\times A$.
For a relation denoted by $\sim$, the notation $a\sim b$ means that
$(a,b)$ belongs to that subset.  The relation is \emph{reflexive} if
$a\sim a$ for every $a\in A$; \emph{symmetric} if
$a\sim b\Rightarrow b\sim a$ for every $a,b\in A$; and
\emph{transitive} if $a\sim b$ and $b\sim c$ imply $a\sim c$ for every
$a,b,c\in A$.  An \emph{equivalence relation} has all three properties.
If $a\in A$, its \emph{equivalence class} is
$[a]=\{x\in A:x\sim a\}$.  A \emph{partition} of $A$ is a family of
non-empty subsets $(A_i)_{i\in I}$, where $I$ is a set, whose union is $A$
and which are \emph{pairwise disjoint}: $A_i\cap A_j=\varnothing$ whenever
$i,j\in I$ and $i\ne j$.

\examquestion{\label{q:equivalence}
Let $\sim$ be an equivalence relation on a set $A$, and, for every
$a\in A$, write $[a]=\{x\in A:x\sim a\}$.
\begin{subquestions}
  \item \nameditem{EEC}{Equivalent Elements have the same Class}
  For all $a,b\in A$, prove that $a\sim b$ if and only if
  $[a]=[b]$.
  \item \nameditem{ECED}{Equivalence Classes are Equal or Disjoint}
  Deduce that two equivalence classes are either equal or disjoint.
  \item \nameditem{ECPS}{Equivalence Classes Partition the Set}
  Prove that the equivalence classes form a partition of $A$.
  \item \nameditem{PDER}{Partitions Define Equivalence Relations}
  Conversely, let $I$ be a set and let $(A_i)_{i\in I}$ be a
  partition of $A$.  Define a relation whose equivalence classes are
  precisely the sets $A_i$.
\end{subquestions}}

\concept{Quotients}
If $\sim$ is an equivalence relation on a set $A$, its \emph{quotient set}
is
\[
  A/{\sim}=\{[a]:a\in A\}.
\]
The \emph{quotient map} is $q\colon A\to A/{\sim}$ defined by
$q(a)=[a]$ for every $a\in A$.  A formula defining a function on
equivalence classes is \emph{well-defined} if its value does not depend on
the representative chosen for the class.

\examquestion{\label{q:quotient}
Retain the set $A$ and the equivalence relation $\sim$ from Question
\ref{q:equivalence}.  Let $B$ be a set, let $q\colon A\to A/{\sim}$ be the
quotient map, and let $f\colon A\to B$ be a function.
\begin{subquestions}
  \item \nameditem{QMS}{Quotient Maps are Surjective}
  Show that $q$ is surjective.
  \item \nameditem{FQWD}{Factorization through Quotients is Well-Defined}
  Suppose that, for all $a,a'\in A$, the relation $a\sim a'$ implies
  $f(a)=f(a')$.  Construct a map
  $\bar f\colon A/{\sim}\to B$ such that $f=\bar f\circ q$, and prove that it
  is well-defined.
  \item \nameditem{FQU}{Factorization through Quotients is Unique}
  Prove that $\bar f$ is unique.
  \item \nameditem{FCC}{Factorization and Constancy on Classes}
  Prove the converse: if $f=\bar f\circ q$ for some function
  $\bar f\colon A/{\sim}\to B$, then $f$ is constant on every equivalence
  class.
\end{subquestions}
\indication{The only possible definition is $\bar f([a])=f(a)$.  The main
point is to show that the answer does not depend on the representative.}}

\concept{Power sets and orders}
For a set $X$, its \emph{power set} is
$\mathcal P(X)=\{A:A\subseteq X\}$.  A relation $\preccurlyeq$ on a set
$Y$ is a \emph{partial order} if, for all $x,y,z\in Y$, it is
\begin{properties}
  \item reflexive: $x\preccurlyeq x$;
  \item antisymmetric: $x\preccurlyeq y$ and $y\preccurlyeq x$ imply
  $x=y$;
  \item transitive: $x\preccurlyeq y$ and $y\preccurlyeq z$ imply
  $x\preccurlyeq z$.
\end{properties}
It is a \emph{total order} if, in addition, either $x\preccurlyeq y$ or
$y\preccurlyeq x$ for every $x,y\in Y$.

For $a,b\in\Z$, write $a\mid b$ and say that $a$ \emph{divides} $b$ if
$b=ak$ for some $k\in\Z$.  For $m\in\N$, put
\[
  m\Z=\{mk:k\in\Z\},\qquad
  a+m\Z=\{a+mk:k\in\Z\}\quad(a\in\Z).
\]
The latter set is called the \emph{coset} of $m\Z$ represented by $a$.

\examquestion{\label{q:orders}
Let $X$ be a set.
\begin{subquestions}
  \item \nameditem{IOPS}{Inclusion Orders the Power Set}
  Show that inclusion is a partial order on $\mathcal P(X)$.  Determine
  exactly for which sets $X$ this order is total.
  \item \nameditem{DPO}{Divisibility as a Partial Order}
  Show that divisibility is a partial order on $\N$ but not on $\Z$.
  Which axiom fails on $\Z$?
  \item \nameditem{CER}{Congruence is an Equivalence Relation}
  For $m\in\N$ and $a,b\in\Z$, define $a\sim_m b$ by
  $m\mid(a-b)$.  Prove that this is an equivalence relation on $\Z$ and
  show that the equivalence class of every $a\in\Z$ is $a+m\Z$.
\end{subquestions}}

\exampart{II}{Induction and finite encoding}

This part develops induction, well-ordering, and the principal tools for
counting finite structures.

\subpart{A. --- Induction and descent}

Write $\N_0=\{0,1,2,\ldots\}$.

\examquestion{\label{q:peano-induction}
Assume the following subset form of the Peano induction axiom: if
$S\subseteq\N_0$, $0\in S$, and, for every $n\in\N_0$, membership
$n\in S$ implies $n+1\in S$, then $S=\N_0$.
\begin{subquestions}
  \item \nameditem{WIPA}{Weak Induction from Peano Axioms}
  For every $n\in\N_0$, let $P(n)$ be a statement.  Prove the
  \emph{weak induction principle}: if $P(0)$ is true and
  $P(n)\Rightarrow P(n+1)$ for every $n\in\N_0$, then $P(n)$ is true
  for every $n\in\N_0$.
  \item \nameditem{ISO}{Induction Starting at One}
  For every $n\in\N$, let $Q(n)$ be a statement.  Prove the
  corresponding principle with base case $Q(1)$ and induction step
  $Q(n)\Rightarrow Q(n+1)$ for every $n\in\N$.
\end{subquestions}}

\examquestion{\label{q:strong-induction}
Let $P(n)$ be a statement for every $n\in\N$.  The strong induction
principle says that, if
\[
  \bigl[(\forall k\in\N\text{ with }k<n)\ P(k)\bigr]\Longrightarrow P(n)
  \quad\text{for every }n\in\N,
\]
then $P(n)$ holds for every $n\in\N$.
\begin{subquestions}
  \item \nameditem{SIIWI}{Strong Induction Implies Weak Induction}
  Show that strong induction implies weak induction.
  \item \nameditem{WIPSI}{Weak Induction Proves Strong Induction}
  Assuming weak induction, for each $n\in\N$ let
  $Q(n)$ be the assertion that $P(k)$ holds for every $k\in\N$ with
  $k\leq n$.
  Prove strong induction.
  \item \nameditem{SICBC}{Strong Induction Contains its Base Case}
  Explain why the implication defining strong induction already
  contains its base case.
\end{subquestions}
\indication{For part (b), prove $Q(1)$ and, for every $n\in\N$, prove
$Q(n)\Rightarrow Q(n+1)$.}}

\concept{Least elements and well-orders}
Let $Y$ be a set with a total order $\preccurlyeq$, and let
$S\subseteq Y$.  An element $m\in S$ is a \emph{least element} of $S$ if
$m\preccurlyeq s$ for every $s\in S$.  The order is a \emph{well-order}
if every non-empty subset of $Y$ has a least element.

\examquestion{\label{q:well-ordering}
The well-ordering principle asserts that every non-empty subset of $\N$ has
a least element.
\begin{subquestions}
  \item \nameditem{WONI}{Well-Ordering of Non-negative Integers}
  Deduce that every non-empty subset of $\N_0$ also has a least
  element.
  \item \nameditem{SIWO}{Strong Induction from Well-Ordering}
  Assuming well-ordering, prove strong induction by considering the
  least counterexample.
  \item \nameditem{WOSI}{Well-Ordering from Strong Induction}
  Assuming strong induction, prove well-ordering.
  \item \nameditem{PPLC}{Prime Products by Least Counterexample}
  Use a least-counterexample argument to prove that every
  $n\in\N$ with $n\geq2$ is a product of primes.
\end{subquestions}
\indication{For part (c), if a non-empty set $S$ had no least element,
consider the assertion that none of $1,\ldots,n$ belongs to $S$.}}

\admitted{WFRT}{Well-Founded Recursion Theorem}{Let $Y$ be a set with a
well-order $\preccurlyeq$, let $X$ be a set, and, for every $y\in Y$, put
$Y_{<y}=\{z\in Y:z\preccurlyeq y\text{ and }z\ne y\}$.  Suppose that a rule
$F$ assigns an element $F(y,g)\in X$ to every $y\in Y$ and every map
$g\colon Y_{<y}\to X$.  Then there is a unique map $f\colon Y\to X$ such
that $f(y)=F(y,f|_{Y_{<y}})$ for every $y\in Y$, where
$f|_{Y_{<y}}\colon Y_{<y}\to X$ is the restriction of $f$ to $Y_{<y}$.}

\examquestion{\label{q:lexicographic}
Write $\N_0^2=\N_0\times\N_0$.  For
$(a,b),(c,d)\in\N_0^2$, define the lexicographic order by
\[
  (a,b)\leq_{\mathrm{lex}}(c,d)
  \quad\Longleftrightarrow\quad
  a<c\ \text{ or }\ (a=c\text{ and }b\leq d).
\]
Write $(a,b)<_{\mathrm{lex}}(c,d)$ when
$(a,b)\leq_{\mathrm{lex}}(c,d)$ and $(a,b)\ne(c,d)$.
\begin{subquestions}
  \item \nameditem{LWOP}{Lexicographic Well-Ordering of Pairs}
  Prove that $\leq_{\mathrm{lex}}$ is a well-order on $\N_0^2$.
  \item \nameditem{ARWF}{Ackermann Recursion is Well-Founded}
  For $m,n\in\N_0$, the Ackermann function
  $A\colon\N_0^2\to\N_0$ is specified recursively by
  \[
  A(m,n)=
  \begin{cases}
    n+1,&m=0,\\
    A(m-1,1),&m>0,\ n=0,\\
    A(m-1,A(m,n-1)),&m>0,\ n>0.
  \end{cases}
  \]
  Show that every recursive call on the right has lexicographically smaller
  first two input coordinates than $(m,n)$ at the moment it is needed.
  Explain why well-ordering makes the specification legitimate.
\end{subquestions}}

\examquestion{\label{q:hanoi}
For each $n\in\N_0$, consider $n$ discs of distinct sizes.  In the Tower of
Hanoi, these discs must be moved between three pegs, one disc at a time,
with no larger disc ever placed on a smaller one.  Let $M_n\subseteq\N_0$
be the set of numbers of moves used by legal strategies which transfer the
complete tower from one specified peg to another.
\begin{subquestions}
  \item \nameditem{EOHS}{Existence of an Optimal Hanoi Strategy}
  Prove by induction on $n\in\N_0$ that $M_n$ is non-empty.  By
  well-ordering, it therefore has a least element; denote this element by
  $T_n\in\N_0$.  Prove that $T_0=0$.
  \item \nameditem{HRUB}{Hanoi Recurrence Upper Bound}
  For every $n\in\N$, construct a strategy showing
  $T_n\leq2T_{n-1}+1$.
  \item \nameditem{HRLB}{Hanoi Recurrence Lower Bound}
  For every $n\in\N$, prove that every legal strategy requires at
  least $2T_{n-1}+1$ moves, and hence that
  $T_n\geq2T_{n-1}+1$.
  \item \nameditem{EHMC}{Exact Hanoi Move Count}
  Deduce by induction that $T_n=2^n-1$ for every $n\in\N_0$.
\end{subquestions}}

\concept{Finite sets and full binary trees}
For $n\in\N$, put $[n]=\{1,2,\ldots,n\}$, and put
$[0]=\varnothing$.  A set $A$ is \emph{finite} if there is a bijection
from $A$ to $[n]$ for some $n\in\N_0$.

We use the following recursive description.  A single vertex is a finite
rooted full binary tree, and that vertex is its root.  Given two such trees,
join their roots to a new root; the two old roots are the \emph{children}
of the new root, and each joining line is an \emph{edge}.  Every tree under
consideration is obtained by finitely many such steps.  A vertex with no
children is a \emph{leaf}; a vertex with two children is an \emph{internal
vertex}.

\examquestion{\label{q:binary-trees}
Let $T$ be a finite rooted full binary tree.  Let $L(T)\in\N$ and
$I(T)\in\N_0$ denote its numbers of leaves and internal vertices.
\begin{subquestions}
  \item \nameditem{FBTVFI}{Full Binary Tree Vertex Formula by Induction}
  Prove by strong induction on $L(T)$ that
  $I(T)=L(T)-1$.
  \item \nameditem{FBTFEC}{Full Binary Tree Formula by Edge Counting}
  Give a second proof by counting edges in two ways.
\end{subquestions}}
\indication{For part (b), if $E(T)\in\N_0$ is the number of edges, prove
from the recursive construction that
$E(T)=L(T)+I(T)-1$, and also observe that $E(T)=2I(T)$.}

\subpart{B. --- Finite sets and counting}

\examquestion{\label{q:finite-injection}
Let $n\in\N_0$.
\begin{subquestions}
  \item \nameditem{FSIS}{Finite Self-Injection is Surjective}
  Prove by induction that every injection $f\colon[n]\to[n]$ is
  surjective.
  \item \nameditem{FSSI}{Finite Self-Surjection is Injective}
  Deduce that every surjection $[n]\to[n]$ is injective.
\end{subquestions}}
\indication{A transposition of $[n]$ is a bijection $[n]\to[n]$ which
swaps two specified elements and fixes all others.  For part (a), when $n\in\N$,
compose $f$ with a transposition so that the resulting map fixes $n$, then
restrict it to $[n-1]$.  For part (b), given a surjection
$f\colon[n]\to[n]$, choose one pre-image of each element to obtain a
function $g\colon[n]\to[n]$ with $f\circ g=\operatorname{id}_{[n]}$; apply
part (a) to $g$.}

\concept{Cardinality}
If a finite set $E$ admits a bijection to $[n]$, where $n\in\N_0$, we
provisionally write $|E|=n$ and call this number the \emph{cardinality} or
\emph{size} of $E$.  Part (a) of the next question proves that this notation
does not depend on the chosen bijection.

\examquestion{\label{q:finite-cardinality}
Let $m,n\in\N_0$.
\begin{subquestions}
  \item \nameditem{UFC}{Uniqueness of Finite Cardinality}
  Suppose that a set $A$ admits bijections to both $[m]$ and $[n]$.
  Use Question \ref{q:finite-injection} to prove $m=n$.
  \item \nameditem{FPP}{Finite Pigeonhole Principle}
  Let $A$ and $B$ be finite sets, and let $f\colon A\to B$ be a
  function with
  $|A|>|B|$.  Prove that $f$ is not injective.
  \item \nameditem{GFPP}{Generalized Finite Pigeonhole Principle}
  Assume in addition that $n\in\N$.  Let
  $f\colon[mn+1]\to[n]$ be a function, interpreted as placing object
  $i\in[mn+1]$ in box $f(i)\in[n]$.  Prove that there exists $j\in[n]$
  for which the pre-image $f^{-1}(\{j\})$ has at least $m+1$ elements.
\end{subquestions}}
\indication{For part (a), suppose for example that $m<n$.  Use the
bijections and the function $[m]\to[n]$ defined by $i\mapsto i$ to
construct an injective but
non-surjective function $[n]\to[n]$, contradicting Question
\ref{q:finite-injection}.  For part (c), if every fibre had at most $m$
elements, number the members of each fibre using distinct labels in
$[m]$.  This would give an injection $[mn+1]\to[m]\times[n]$; identify
$[m]\times[n]$ with $[mn]$ by an explicit bijection.}

The conclusion of Question \ref{q:finite-cardinality}(c) is called the
\emph{generalised pigeonhole principle}; its case $m=1$ is the
\emph{pigeonhole principle}.

\concept{Permutations, factorials, and binomial coefficients}
A \emph{permutation} of a set $E$ is a bijection $E\to E$.  For
$s\in\N$, define $s!=1\cdot2\cdots s$, and define $0!=1$.  For
$n\in\N_0$ and $s\in\Z$, the \emph{binomial coefficient} $\binom ns$ is
the number of $s$-element subsets of $[n]$ when $0\leq s\leq n$, and is
$0$ otherwise.  More generally, a product with no factors has value $1$,
and a sum with no terms has value $0$.

\examquestion{\label{q:counting-maps}
Let $n,r\in\N_0$ satisfy $0\leq r\leq n$.
For this question, use the counting convention $0^0=1$.
\begin{subquestions}
  \item \nameditem{CFFS}{Counting Functions between Finite Sets}
  Count the maps $[r]\to[n]$.
  \item \nameditem{CIP}{Counting Injections and Permutations}
  Count the injections $[r]\to[n]$ and hence the permutations of
  $[n]$.
  \item \nameditem{BCFF}{Binomial Coefficient Factorial Formula}
  By counting the orderings of each $r$-element subset, prove
  \[
    \binom nr=\frac{n!}{r!(n-r)!}.
  \]
  \item \nameditem{SBC}{Sum of Binomial Coefficients}
  Deduce that $\sum_{j=0}^n\binom nj=2^n$.
\end{subquestions}}
\indication{First justify the multiplication principle by induction: if a
choice is made in $r\in\N_0$ successive stages and stage $i\in[r]$ has
exactly $c_i\in\N_0$ possible outcomes after every fixed set of earlier
choices, then the number of complete outcomes is
$\prod_{i=1}^r c_i$.  Apply it by choosing the image of each domain element
in turn.}

\concept{Indicator functions}
Let $X$ be a set and let $A\subseteq X$.  The \emph{indicator function} of
$A$ is the function $\mathbf1_A\colon X\to\{0,1\}$ defined, for every
$x\in X$, by
\[
  \mathbf1_A(x)=
  \begin{cases}
    1,&x\in A,\\
    0,&x\notin A.
  \end{cases}
\]
Expressions involving indicator functions are evaluated pointwise; in such
an expression, $1$ denotes the function which has value $1$ at every
$x\in X$.  More explicitly, if $u,v\colon X\to\Z$, then
$(u+v)(x)=u(x)+v(x)$, $(u-v)(x)=u(x)-v(x)$, and
$(uv)(x)=u(x)v(x)$ for every $x\in X$.

\examquestion{\label{q:indicators}
Let $n\in\N$ and let $A,B,A_1,\ldots,A_n$ be subsets of a finite set $X$.
\begin{subquestions}
  \item \nameditem{IIBSO}{Indicator Identities for Basic Set Operations}
  Prove
  \[
    \mathbf1_{X\setminus A}=1-\mathbf1_A,\qquad
    \mathbf1_{A\cap B}=\mathbf1_A\mathbf1_B,\qquad
    \mathbf1_{A\cup B}=\mathbf1_A+\mathbf1_B-\mathbf1_A\mathbf1_B.
  \]
  \item \nameditem{IFSD}{Indicator Formula for Symmetric Difference}
  Express $\mathbf1_{A\triangle B}$ in terms of
  $\mathbf1_A,\mathbf1_B$, and use it to give another proof that symmetric
  difference is associative.
  \item \nameditem{IEIE}{Inclusion Exclusion by Indicator Expansion}
  Expand $\prod_{i=1}^n(1-\mathbf1_{A_i})$ and sum over $X$ to prove
  the inclusion--exclusion formula
  \[
    \left|\bigcap_{i=1}^n(X\setminus A_i)\right|
    =\sum_{S\subseteq[n]}(-1)^{|S|}
      \left|\bigcap_{i\in S}A_i\right|,
  \]
  where the intersection indexed by $\varnothing$ is $X$.
\end{subquestions}
\indication{Use $|A|=\sum_{x\in X}\mathbf1_A(x)$ only after completing the
pointwise product expansion.}}

\examquestion{\label{q:binomial}
Let $n,r\in\N_0$ satisfy $0\leq r\leq n$.
\begin{subquestions}
  \item \nameditem{BTPE}{Binomial Theorem by Product Expansion}
  For all $x,y\in\R$, prove the binomial theorem
  \[
    (x+y)^n=\sum_{j=0}^{n}\binom njx^jy^{n-j}
  \]
  by expanding a product of $n$ factors.
  \item \nameditem{BCSC}{Binomial Coefficient Symmetry from Complements}
  Prove the symmetry
  $\binom nr=\binom n{n-r}$ by taking complements.
  \item \nameditem{PISP}{Pascal Identity by Subset Partition}
  Partition the $r$-element subsets of $[n+1]$ according to whether
  they contain $n+1$ and deduce Pascal's identity
  \[
    \binom{n+1}{r}=\binom nr+\binom n{r-1}.
  \]
  \item \nameditem{PIF}{Pascal Identity from Factorials}
  When $1\leq r\leq n$, derive the same identity algebraically from
  the factorial formula in Question \ref{q:counting-maps}.
\end{subquestions}}

\concept{Falling factorials}
For $x\in\R$ and $s\in\N$, define
$x^{\underline s}=x(x-1)\cdots(x-s+1)$, and define
$x^{\underline0}=1$.

\examquestion{\label{q:double-counting}
Let $n,a,b,r,k\in\N_0$.  In part (a), assume
$0\leq r\leq k\leq n$; in parts (b) and (c), assume $r\leq a+b$.
\begin{subquestions}
  \item \nameditem{NSDC}{Nested Subset Double Counting}
  By counting pairs $(R,K)$ with
  $R\subseteq K\subseteq[n]$, $|R|=r$, and $|K|=k$, prove
  \[
    \binom nk\binom kr=\binom nr\binom{n-r}{k-r}.
  \]
  \item \nameditem{VIDC}{Vandermonde Identity by Double Counting}
  Let $Y$ be a finite set partitioned into subsets $Y_1,Y_2$ with
  $|Y_1|=a$ and $|Y_2|=b$.  Count its
  $r$-element subsets according to how many elements they take from the
  first part, and deduce Vandermonde's identity
  \[
    \sum_{j=0}^r\binom aj\binom b{r-j}=\binom{a+b}{r}.
  \]
  \item \nameditem{FFVI}{Falling Factorial Vandermonde Identity}
  Derive
  \[
    (a+b)^{\underline r}
      =\sum_{j=0}^r\binom rj
        a^{\underline j}b^{\underline{r-j}}
  \]
  for non-negative integers $a,b$.
\end{subquestions}}

\concept{Tuples and words}
For $n\in\N$, write $\N_0^n$ for the set of ordered $n$-tuples of elements
of $\N_0$.  For $k\in\N_0$, a \emph{word of length $k$} in the two
symbols $\star$ and $\vert$ is a function
$w\colon[k]\to\{\star,\vert\}$.  Its entries are
$w(1),\ldots,w(k)$ when $k\in\N$; the word of length $0$ has no entries.
The symbols are called a star and a bar, respectively.  Such a word
\emph{contains $r$ stars and $s$ bars}, where $r,s\in\N_0$, if
$|w^{-1}(\{\star\})|=r$ and $|w^{-1}(\{\vert\})|=s$.

\examquestion{\label{q:stars-bars}
Let $n\in\N$ and $r\in\N_0$.
\begin{subquestions}
  \item \nameditem{SBE}{Star-Bar Encoding}
  Construct a bijection between the tuples
  $(x_1,\ldots,x_n)\in\N_0^n$ satisfying $x_1+\cdots+x_n=r$ and words
  containing $r$ stars and $n-1$ bars.
  \item \nameditem{WCCF}{Weak Composition Counting Formula}
  Deduce that the number of such solutions is
  $\binom{n+r-1}{r}$.
  \item \nameditem{SCIE}{Surjection Count by Inclusion Exclusion}
  Assuming $r\geq1$, use inclusion--exclusion to show that the number
  of surjections
  $[r]\to[n]$ is
  \[
    \sum_{j=0}^{n}(-1)^j\binom nj(n-j)^r.
  \]
\end{subquestions}}

\examquestion{\label{q:derangements}
For $n\in\N_0$, a \emph{fixed point} of a permutation
$\sigma\colon[n]\to[n]$ is an element $i\in[n]$ such that
$\sigma(i)=i$.  Let $D_n$ be the number of permutations of $[n]$ without a
fixed point; in particular, $D_0=1$.
\begin{subquestions}
  \item \nameditem{DFIE}{Derangement Formula by Inclusion Exclusion}
  Apply inclusion--exclusion to prove
  \[
    D_n=n!\sum_{j=0}^{n}\frac{(-1)^j}{j!}.
  \]
  \item \nameditem{DRF}{Derangement Recurrence Formula}
  Prove the recurrence
  \[
    D_n=(n-1)(D_{n-1}+D_{n-2})\qquad(n\geq2).
  \]
\end{subquestions}
\indication{For part (b), let $\sigma\colon[n]\to[n]$ be a derangement,
put $j=\sigma(1)\in[n]$, and separate the cases $\sigma(j)=1$ and
$\sigma(j)\ne1$.  In the latter case, let $k\in[n]$ be the unique element
with $\sigma(k)=1$.  Remove $1$ by defining a permutation $\tau$ of
$[n]\setminus\{1\}$ by $\tau(k)=j$ and $\tau(i)=\sigma(i)$ for every
$i\in[n]\setminus\{1,k\}$.  Reverse this construction by inserting $1$
immediately before $j$.}}

\exampart{III}{Euclidean arithmetic and prime factorisation}

This part uses well-ordering to construct quotients and greatest common
divisors.  It then develops the arithmetic of primes from these constructions.

\subpart{A. --- Division and the Euclidean algorithm}

\examquestion{\label{q:division}
Let $a\in\Z$ and $b\in\N$.  Consider
\[
  S=\{a-kb:k\in\Z,\ a-kb\geq0\}.
\]
\begin{subquestions}
  \item \nameditem{LRE}{Least Remainder Existence}
  Show that $S$ is a non-empty subset of $\N_0$ and therefore has a
  least element $r\in\N_0$.
  \item \nameditem{DRB}{Division Remainder Bounds}
  Since $r\in S$, there is a $q\in\Z$ such that $r=a-qb$.
  Prove that $0\leq r<b$.
  \item \nameditem{QRU}{Quotient and Remainder Uniqueness}
  Suppose that $q',r'\in\Z$ also satisfy
  $a=q'b+r'$ and $0\leq r'<b$.  Prove that $q'=q$ and $r'=r$.
  \item \nameditem{SEDE}{Signed Euclidean Division Example}
  Find these integers $q$ and $r$ when $a=-137$ and $b=12$.
\end{subquestions}
\indication{If the least element satisfied $r\geq b$, subtracting one more
copy of $b$ would produce a smaller member of $S$.  To prove that $S$ is
non-empty, take $k=0$ if $a\geq0$, and take $k=a$ if $a<0$.}}

The conclusion of Question \ref{q:division} is called the \emph{Division
Algorithm}.  In the unique expression $a=qb+r$ with $0\leq r<b$, the
integers $q$ and $r$ are called the \emph{quotient} and \emph{remainder}.

\concept{Common divisors and greatest common divisors}
Let $a,b\in\N$.  An integer $c\in\Z$ is a \emph{common divisor} of $a$ and
$b$ if $c\mid a$ and $c\mid b$.  A \emph{greatest common divisor} of $a$
and $b$ is a number $d\in\N$ such that $d\mid a$, $d\mid b$, and every
common divisor of $a$ and $b$ divides $d$.  If it exists, it is unique and
is denoted by $\gcd(a,b)$.

\examquestion{\label{q:gcd-construction}
Let $a,b\in\N$ and put
\[
  L=\{ua+vb:u,v\in\Z,\ ua+vb>0\}.
\]
\begin{subquestions}
  \item \nameditem{LPLC}{Least Positive Linear Combination}
  Show that $L$ is a non-empty subset of $\N$ and therefore has a
  least element; denote it by $d\in\N$.
  \item \nameditem{LLCDI}{Least Linear Combination Divides both Inputs}
  Divide $a$ by $d$ and use the minimality of $d$ to prove $d\mid a$.
  Prove similarly that $d\mid b$.
  \item \nameditem{CDDM}{Common Divisors Divide the Minimum}
  Show that every $c\in\Z$ which is a common divisor of $a$ and $b$
  divides $d$.
  \item \nameditem{GCDBI}{Greatest Common Divisor and Bezout Identity}
  Deduce that $d$ is the unique greatest common divisor of $a,b$ and
  prove B\'ezout's identity:
  \[
    \gcd(a,b)=ua+vb
  \]
  for some $u,v\in\Z$.
  \item \nameditem{GCDZS}{Greatest Common Divisor with Zero and Signs}
  For $z\in\Z$, define $|z|=z$ if $z\geq0$ and $|z|=-z$ if $z<0$.
  For $s,t\in\Z$ which are not both zero, define
  \[
    \gcd(s,t)=
    \begin{cases}
      \gcd(|s|,|t|),&s\ne0\text{ and }t\ne0,\\
      |s|,&t=0,\\
      |t|,&s=0.
    \end{cases}
  \]
  Prove that this number divides both $s$ and $t$, is divisible by every
  common divisor of $s,t$, and satisfies
  $\gcd(s,t)=us+vt$ for some $u,v\in\Z$.
\end{subquestions}}

\examquestion{\label{q:euclid}
Let $a,b\in\N$ satisfy $a\geq b$.  Put $r_{-1}=a$ and $r_0=b$.  For every
$j\in\N$ for which $r_{j-1}>0$, apply the Division Algorithm to define
$q_j\in\N$ and $r_j\in\N_0$ by
\[
  r_{j-2}=q_jr_{j-1}+r_j,\quad 0\leq r_j<r_{j-1}.
\]
\begin{subquestions}
  \item \nameditem{CDPR}{Common Divisors are Preserved by Remainders}
  For every $j\in\N$ for which $r_j$ is defined, show that
  $r_{j-2}$ and $r_{j-1}$ have exactly the same common divisors as
  $r_{j-1}$ and $r_j$.
  \item \nameditem{SDT}{Strict Descent and Termination}
  Prove that the successive non-zero remainders lie in $\N$ and
  strictly decrease.  Deduce that only finitely many non-zero remainders
  occur, so the algorithm terminates.
  \item \nameditem{LNRGCD}{Last Non-zero Remainder is the Greatest Common Divisor}
  Prove that the last non-zero remainder is $\gcd(a,b)$.
\end{subquestions}}

\concept{A logarithmic bound}
For $b\in\N$, provisionally let $\lambda(b)$ be the least $t\in\N$ such
that $b<2^t$.  Part (a) of the next question proves that this integer
exists.  We say that the number of divisions is $O(\log b)$ if there is a
constant $C\in\N$, independent of $a$ and $b$, such that the algorithm
uses at most $C\lambda(b)$ divisions whenever $a,b\in\N$ and $a\geq b$.

\examquestion{\label{q:euclid-complexity}
Retain the notation of Question \ref{q:euclid}.
\begin{subquestions}
  \item \nameditem{BLWD}{Binary Length is Well-Defined}
  Prove by induction that $2^b>b$ for every $b\in\N$, and deduce
  from well-ordering that $\lambda(b)$ is well-defined.
  \item \nameditem{TESH}{Two Euclidean Steps Halve}
  For every $j\in\N_0$ for which $r_{j-1},r_j,r_{j+1}$ are defined
  and non-zero, prove that $r_{j+1}<\frac12r_{j-1}$.
  \item \nameditem{LEDB}{Logarithmic Euclidean Division Bound}
  Deduce that Euclid's algorithm uses at most
  $2\lambda(b)+1$ divisions, and hence uses $O(\log b)$ divisions.
\end{subquestions}
\indication{For part (b), if $r_j\leq r_{j-1}/2$ there is nothing to prove.
Otherwise
the next quotient is at least $1$ and
$r_{j+1}\leq r_{j-1}-r_j<r_{j-1}/2$.}}

\examquestion{\label{q:extended-euclid}
With the notation of Question \ref{q:euclid}, define
\[
  (x_{-1},y_{-1})=(1,0),\qquad (x_0,y_0)=(0,1),
\]
and, for every $j\in\N$ for which $q_j$ is defined,
\[
  x_j=x_{j-2}-q_jx_{j-1},\qquad
  y_j=y_{j-2}-q_jy_{j-1}.
\]
\begin{subquestions}
  \item \nameditem{EELCI}{Extended Euclidean Linear Combination Invariant}
  For every integer $j\geq-1$ for which $r_j$ is defined, prove by
  induction that $r_j=x_ja+y_jb$.
  \item \nameditem{EEDI}{Extended Euclidean Determinant Identity}
  For every $j\in\N$ for which $x_j,y_j$ are defined, show that
  \[
    x_jy_{j-1}-x_{j-1}y_j=(-1)^{j+1}.
  \]
  Deduce in particular that $\gcd(x_j,y_j)=1$.
  \item \nameditem{EBREE}{Explicit Bezout Relation by Extended Euclid}
  Apply the algorithm to $a=2026$ and $b=748$, and give an explicit
  B\'ezout relation.
  \item \nameditem{EQCFC}{Euclidean Quotients and Continued Fraction Convergents}
  Define $[c_0]=c_0$ for every $c_0\in\Z$.  For $t\in\N$,
  $c_0\in\Z$, and $c_1,\ldots,c_t\in\N$, define
  \[
    [c_0;c_1,\ldots,c_t]
      =c_0+\cfrac1{c_1+\cfrac1{\ddots+\cfrac1{c_t}}}.
  \]
  For $s\in\N_0$ with $s\leq t$, the
  number $[c_0;c_1,\ldots,c_s]$ is called the $s$th \emph{convergent}.
  Read the division
  equations from part (c) backwards to express $2026/748$ as a
  finite continued fraction and list its convergents.
\end{subquestions}}
\indication{For part (b), substitute the recurrences for $x_j$ and $y_j$
to show that the determinant at index $j$ is the negative of the determinant
at index $j-1$.}

\examquestion{\label{q:diophantine}
Write $\Z^2=\Z\times\Z$.  Let $a,b\in\N$, $c\in\Z$, and
$d=\gcd(a,b)$.
\begin{subquestions}
  \item \nameditem{LDSC}{Linear Diophantine Solvability Criterion}
  Prove that $ax+by=c$ has a solution $(x,y)\in\Z^2$ if and only if
  $d\mid c$.
  \item \nameditem{PLDS}{Parametrization of Linear Diophantine Solutions}
  Suppose $d\mid c$ and let $(x_0,y_0)\in\Z^2$ be one solution.
  Put $a'=a/d$ and $b'=b/d$, and prove that $\gcd(a',b')=1$.
  Prove that
  all solutions are
  \[
    x=x_0+\frac bd\,t,\qquad
    y=y_0-\frac ad\,t,\qquad t\in\Z.
  \]
  \item \nameditem{SCLDE}{Solving a Concrete Linear Diophantine Equation}
  Find all integer solutions of $2026x+748y=6$.
\end{subquestions}
\indication{For part (b), divide the equation by $d$.  From
$a'(x-x_0)=-b'(y-y_0)$ and $\gcd(a',b')=1$, use a B\'ezout relation to show
that $a'$ divides $y-y_0$.}}

\subpart{B. --- Prime numbers}

Recall that a natural number $p>1$ is prime if its only positive divisors are
$1$ and $p$.  The prime factorisation existence proved in Question
\ref{q:well-ordering}(d) may be used throughout this subpart.

\examquestion{\label{q:euclid-primes}
Let $k\in\N$, and, for every $i\in[k]$, let $p_i\in\N$ be prime.
\begin{subquestions}
  \item \nameditem{EPALPD}{Euclidean Product Avoids Listed Prime Divisors}
  Show that, for every $i\in[k]$, the prime $p_i$ does not divide
  $p_1\cdots p_k+1$.
  \item \nameditem{EPIP}{Euclid Proof of Infinite Primes}
  Deduce that there are infinitely many primes.
  \item \nameditem{CPEP}{Composite Possibility for the Euclidean Product}
  Explain why the argument does not claim that
  $p_1\cdots p_k+1$ itself is prime.
\end{subquestions}}

\examquestion{\label{q:erdos-primes}
Suppose, for a contradiction, that $k\in\N$ and that
$p_i\in\N$ is prime for every $i\in[k]$ and these are all the primes.
\begin{subquestions}
  \item \nameditem{SDFP}{Squarefree Decomposition over Finite Primes}
  Show that every $n\in\N$ can be written, for some $s\in\N$, as
  \[
    n=s^2p_1^{\varepsilon_1}\cdots p_k^{\varepsilon_k},
    \qquad \varepsilon_i\in\{0,1\}\quad\text{for every }i\in[k].
  \]
  \item \nameditem{CBSR}{Counting Bounded Squarefree Representations}
  Let $H\in\N$.  Prove that an integer $n\in\N$ with $n\leq H^2$
  has an expression of the displayed form with $s\leq H$.  Deduce that at
  most $H2^k$ positive integers can be represented in this way.
  \item \nameditem{ECCIP}{Erdos Counting Contradiction for Infinite Primes}
  Take $H=2^k+1$.  Since $H2^k<H^2$, obtain a contradiction by
  considering the $H^2$ integers in $[H^2]$.  Conclude again that there are
  infinitely many primes.
\end{subquestions}
\indication{In part (a), factor the largest square from the prime
factorisation of $n$.}}

\concept{Coprime integers}
Two integers $a,b\in\Z$ which are not both zero are \emph{coprime} if
$\gcd(a,b)=1$.

\examquestion{\label{q:coprime-divisibility}
Let $a,b,c\in\Z$ with $a\ne0$, let $p\in\N$ be prime, and let
$a_1,\ldots,a_r\in\Z$, where $r\in\N$.
\begin{subquestions}
  \item \nameditem{CDPDOF}{Coprime Divisor of a Product Divides the Other Factor}
  Suppose $\gcd(a,b)=1$ and $a\mid bc$.  Multiply a B\'ezout relation
  for $a,b$ by $c$ to prove $a\mid c$.
  \item \nameditem{ELPD}{Euclid Lemma for Prime Divisors}
  If $p$ is prime and $p\mid ab$, prove that $p\mid a$ or $p\mid b$.
  \item \nameditem{PDFP}{Prime Divisor of a Finite Product}
  Prove that if $p\mid a_1\cdots a_r$, then $p\mid a_i$ for some
  $i\in[r]$.
\end{subquestions}}

The assertion proved in Question \ref{q:coprime-divisibility}(b) is called
\emph{Euclid's lemma}.

\examquestion{\label{q:fta}
Let $r,s\in\N$.  Let $p_i\in\N$ be prime for every $i\in[r]$, and let
$q_j\in\N$ be prime for every $j\in[s]$.  Suppose
\[
  p_1\cdots p_r=q_1\cdots q_s.
\]
\begin{subquestions}
  \item \nameditem{MPFEL}{Matching Prime Factors by Euclid Lemma}
  Prove that $p_1=q_j$ for some $j\in[s]$.
  \item \nameditem{UPF}{Uniqueness of Prime Factorization}
  Cancel equal factors and use induction to prove $r=s$ and, after
  reordering, $p_i=q_i$ for every $i\in[r]$.
  \item \nameditem{UPFT}{Unique Prime Factorization Theorem}
  State the resulting Fundamental Theorem of Arithmetic, including
  both existence and uniqueness.
\end{subquestions}}

\concept{Least common multiples}
For $a,b\in\N$, a \emph{least common multiple} is a number $\ell\in\N$
such that $a\mid\ell$, $b\mid\ell$, and, for every $c\in\N$, if
$a\mid c$ and $b\mid c$, then $\ell\mid c$.  If it exists, it is unique
and is denoted by $\operatorname{lcm}(a,b)$.

\concept{Arithmetic in $\Z[\sqrt{-5}]$}
In this question, $\sqrt{-5}$ is a formal symbol whose square is defined to
be $-5$.  Put
\[
  D=\Z[\sqrt{-5}]=\{u+v\sqrt{-5}:u,v\in\Z\},
\]
where two displayed expressions are equal exactly when their two integer
coefficients are equal.  For $u,v,r,s\in\Z$, define
\[
  (u+v\sqrt{-5})+(r+s\sqrt{-5})=(u+r)+(v+s)\sqrt{-5},
\]
\[
  (u+v\sqrt{-5})(r+s\sqrt{-5})
    =(ur-5vs)+(us+vr)\sqrt{-5}.
\]
Define
$N\colon D\to\N_0$ by $N(u+v\sqrt{-5})=u^2+5v^2$ for all $u,v\in\Z$.
An element $z\in D$ is a \emph{unit} if $zw=1$ for some $w\in D$.
For $z,w\in D$, say that $z$ \emph{divides} $w$ in $D$ if $w=zh$ for
some $h\in D$.  A non-zero non-unit $z\in D$ is \emph{irreducible} if
every factorisation $z=xy$ with $x,y\in D$ has a unit factor.  Two elements
$z,w\in D$ are \emph{associates} if $z=uw$ for some unit $u\in D$.
For $x,y\in\N_0$, the symbols $\min(x,y)$ and $\max(x,y)$
denote respectively the smaller and the larger of them.

\examquestion{\label{q:valuations}
Let $a,b\in\N$.
List the distinct primes occurring in $a$ or $b$ as
$p_1,\ldots,p_t\in\N$, where $t\in\N_0$.  For every $i\in[t]$, let
$\alpha_i,\beta_i\in\N_0$ be such that
\[
  a=\prod_{i=1}^t p_i^{\alpha_i},\qquad
  b=\prod_{i=1}^t p_i^{\beta_i}.
\]
\begin{subquestions}
  \item \nameditem{VFGCDLCM}{Valuation Formulas for Greatest Common Divisor and Least Common Multiple}
  Prove
  \[
    \gcd(a,b)=\prod_{i=1}^t p_i^{\min(\alpha_i,\beta_i)},\qquad
    \operatorname{lcm}(a,b)=
      \prod_{i=1}^t p_i^{\max(\alpha_i,\beta_i)}.
  \]
  \item \nameditem{PGCDLCM}{Product of Greatest Common Divisor and Least Common Multiple}
  Deduce
  $\gcd(a,b)\operatorname{lcm}(a,b)=ab$.
  \item \nameditem{MQN}{Multiplicativity of the Quadratic Norm}
  For every $x,y\in D$, verify that $N(xy)=N(x)N(y)$.
  \item \nameditem{UQIR}{Units in the Quadratic Integer Ring}
  Show that the only units of $D$ are $1$ and $-1$.
  \item \nameditem{IFU}{Irreducible Factorizations are not Unique}
  Use the norm to show that
  $2,3,1-\sqrt{-5},1+\sqrt{-5}$ are irreducible, and explain why
  \[
    6=2\cdot3=(1-\sqrt{-5})(1+\sqrt{-5})
  \]
  displays two factorizations which are inequivalent up to order and
  associates.
  \item \nameditem{FELQIR}{Failure of Euclid Lemma in the Quadratic Integer Ring}
  Show that, in $D$, the element $2$ divides
  $(1-\sqrt{-5})(1+\sqrt{-5})$ but divides neither factor.  Thus identify
  the precise step of Question \ref{q:fta} which fails in the system $D$.
\end{subquestions}
\indication{A factorisation of an element of norm $4$, $6$, or $9$ into
non-units would require elements of norm $2$ or $3$.  Check that neither
equation $u^2+5v^2=2$ nor $u^2+5v^2=3$ has a solution
$(u,v)\in\Z^2$.}}

\exampart{IV}{Residue classes and finite arithmetic}

\subpart{A. --- Congruences and simultaneous congruences}

For integers $a,b\in\Z$ and a positive integer $m\in\N$, we say that
$a$ and $b$ are \emph{congruent modulo $m$}, and write
$a\equiv b\pmod m$, if $m\mid(a-b)$.  Here, for $d,t\in\Z$ with
$d\ne0$, the notation $d\mid t$ means that $t=dk$ for some $k\in\Z$.
For $a\in\Z$, put
\[
  [a]_m=\{b\in\Z:b\equiv a\pmod m\},
  \qquad
  \Z/m\Z=\{[a]_m:a\in\Z\}.
\]
After part (a) below, the sets $[a]_m$ are the equivalence classes of
congruence modulo $m$.

\examquestion{\label{q:congruence}
Fix $m\in\N$.
\begin{subquestions}
  \item \nameditem{CERR}{Congruence is an Equivalence Relation Reproved}
  Prove again, from the definition, that congruence modulo $m$ is an
  equivalence relation on $\Z$.
  \item \nameditem{CRAM}{Congruence Respects Addition and Multiplication}
  Let $a,b,c,d\in\Z$.  If $a\equiv b\pmod m$ and
  $c\equiv d\pmod m$, prove
  \[
    a+c\equiv b+d\pmod m,\qquad ac\equiv bd\pmod m.
  \]
  \item \nameditem{WDARC}{Well-Defined Arithmetic on Residue Classes}
  For $a,b\in\Z$, deduce that
  \[
    [a]_m+[b]_m=[a+b]_m,\qquad
    [a]_m[b]_m=[ab]_m
  \]
  define operations on $\Z/m\Z$ independently of representatives.
  \item \nameditem{RRCR}{Reduction between Residue Class Rings}
  If $e\in\N$ and $e\mid m$, prove that
  \[
    \rho\colon\Z/m\Z\longrightarrow\Z/e\Z,
    \qquad \rho([a]_m)=[a]_e\quad(a\in\Z),
  \]
  is a well-defined surjection.
  \item \nameditem{CLRC}{Complete List of Residue Classes}
  Use the division algorithm to prove that $\Z/m\Z$ has exactly the
  $m$ classes $[0]_m,[1]_m,\ldots,[m-1]_m$: prove both that every class
  appears in this list and that the listed classes are pairwise distinct.
\end{subquestions}}

\examquestion{\label{q:modular-obstruction}
\begin{subquestions}
  \item \nameditem{SCT}{Square Classes modulo Three}
  Determine the set $\{[x^2]_3:x\in\Z\}$ of classes represented by
  squares modulo $3$.
  \item \nameditem{MODE}{Modular Obstruction to a Diophantine Equation}
  Prove that there is no pair $(x,y)\in\Z^2$ satisfying
  \[
    2x^2+3y^3=1.
  \]
  \item \nameditem{TSRSE}{Three Squares do not Represent Seven modulo Eight}
  Determine $\{[x^2]_8:x\in\Z\}$ and prove that there are no
  $x,y,z\in\Z$ for which
  $x^2+y^2+z^2\equiv7\pmod8$.
\end{subquestions}}

\examquestion{\label{q:primes-three-mod-four}
Suppose, for a contradiction, that there are only finitely many primes
congruent to $3$ modulo $4$.  Since $3$ is one such prime, choose
$k\in\N$ and list all of them as $p_1,\ldots,p_k\in\N$.  Set
$N=4p_1\cdots p_k-1\in\N$.
\begin{subquestions}
  \item \nameditem{NNALP}{New Number Avoids Listed Primes}
  Show that $N$ is odd and that $p_i\nmid N$ for every
  $i\in\{1,\ldots,k\}$.
  \item \nameditem{PDCTF}{Prime Divisor Congruent to Three modulo Four}
  Let $M\in\N$ satisfy $M>1$ and $M\equiv3\pmod4$.  Prove that
  $M$ has a prime divisor $q\in\N$ satisfying $q\equiv3\pmod4$.
  \item \nameditem{IMPCTF}{Infinitely Many Primes Congruent to Three modulo Four}
  Obtain a contradiction and conclude that there are infinitely many
  primes congruent to $3$ modulo $4$.
\end{subquestions}}

For $t\in\Z$, define
\[
  |t|=\begin{cases}t,&t\geq0,\\-t,&t<0.\end{cases}
\]
For $t\in\Z$ and $m\in\N$, we use
$\gcd(t,m)$ to mean the positive integer $\gcd(|t|,m)$ constructed in
Question \ref{q:gcd-construction}.  For $u\in\Z$, the class $[u]_m$ is a
\emph{unit modulo $m$} if there is some $v\in\Z$ such that
$uv\equiv1\pmod m$; then $[v]_m$ is called an \emph{inverse} of $[u]_m$.

\examquestion{\label{q:linear-congruence}
Let $a,b\in\Z$, $m\in\N$, and $d=\gcd(a,m)$.
All solutions in this question are integers $x\in\Z$, and two solutions
are identified when they are congruent modulo $m$.
\begin{subquestions}
  \item \nameditem{UECRC}{Units are Exactly Coprime Residue Classes}
  For $u\in\Z$, prove that $[u]_m$ is a unit if and only if
  $\gcd(u,m)=1$.
  \item \nameditem{LCSC}{Linear Congruence Solvability Criterion}
  Prove that $ax\equiv b\pmod m$ is soluble if and only if $d\mid b$.
  \item \nameditem{RSLC}{Reduction of a Solvable Linear Congruence}
  Suppose $d\mid b$ and put
  $a'=a/d$, $b'=b/d$, and $m'=m/d$.  Prove that these are integers, that
  $m'\in\N$, that $\gcd(a',m')=1$, and that
  \[
    ax\equiv b\pmod m
      \quad\Longleftrightarrow\quad
    a'x\equiv b'\pmod {m'}.
  \]
  \item \nameditem{CSCLC}{Complete Solution Classes of a Linear Congruence}
  Suppose $d\mid b$ and let $x_0\in\Z$ be one solution.  Prove that
  the distinct solution classes modulo $m$ are
  \[
    [x_0+jm']_m\qquad(j\in\{0,1,\ldots,d-1\}),
  \]
  so there are exactly $d$ of them.
  \item \nameditem{CLCS}{Concrete Linear Congruence Solution}
  Solve $84x\equiv30\pmod{138}$.
\end{subquestions}}

\examquestion{\label{q:crt}
Let $m,n\in\N$ satisfy $\gcd(m,n)=1$, and let $a,b\in\Z$.
\begin{subquestions}
  \item \nameditem{TMCRC}{Two-Modulus Chinese Remainder Construction}
  Choose $u,v\in\Z$ with $um+vn=1$ and verify that
  \[
    x=b\,um+a\,vn
  \]
  satisfies $x\equiv a\pmod m$ and $x\equiv b\pmod n$.
  \item \nameditem{CRUP}{Chinese Remainder Uniqueness modulo the Product}
  Prove that any two simultaneous solutions are congruent modulo
  $mn$: that is, if $x,y\in\Z$ both satisfy the two congruences, prove
  $x\equiv y\pmod{mn}$.
  \item \nameditem{FCRT}{Finite Chinese Remainder Theorem}
  Let $r\in\N$, and let $m_1,\ldots,m_r\in\N$ be
  \emph{pairwise coprime}, meaning that, for all distinct $i,j\in[r]$,
  \[
    \gcd(m_i,m_j)=1.
  \]
  Let
  $a_1,\ldots,a_r\in\Z$, and put
  $M=\prod_{i=1}^r m_i$.  Prove by induction on $r$ that there is an
  $x\in\Z$ satisfying
  \[
    x\equiv a_i\pmod{m_i}
    \qquad\text{for every }i\in\{1,\ldots,r\},
  \]
  and that any two such integers are congruent modulo $M$.
  \item \nameditem{CRNE}{Chinese Remainder Numerical Example}
  Find the least $x\in\N_0$ satisfying
  \[
    x\equiv2\pmod7,\qquad x\equiv3\pmod8,\qquad x\equiv4\pmod9.
  \]
\end{subquestions}}

The existence and uniqueness assertion proved in Question \ref{q:crt}(c)
is called the \emph{Chinese remainder theorem}.

If $m,n\in\N$, addition and multiplication on
$\Z/m\Z\times\Z/n\Z$ are defined coordinate by coordinate.

\examquestion{\label{q:crt-map}
Let $m,n\in\N$, assume $\gcd(m,n)=1$, and consider
\[
  \Phi\colon\Z/mn\Z\longrightarrow\Z/m\Z\times\Z/n\Z,\qquad
  \Phi([x]_{mn})=([x]_m,[x]_n)\quad(x\in\Z).
\]
\begin{subquestions}
  \item \nameditem{CRMB}{Chinese Remainder Map is a Bijection}
  Prove that $\Phi$ is a well-defined bijection.
  \item \nameditem{CRMPRO}{Chinese Remainder Map Preserves Ring Operations}
  For all $x,y\in\Z$, prove
  \[
    \Phi([x]_{mn}+[y]_{mn})
      =\Phi([x]_{mn})+\Phi([y]_{mn}),
  \]
  and prove the analogous equality for multiplication.
  \item \nameditem{UCRD}{Units under Chinese Remainder Decomposition}
  For $c\in\Z$, deduce that $[c]_{mn}$ is a unit if and only if both
  $[c]_m$ and $[c]_n$ are units.
\end{subquestions}}

\subpart{B. --- Units, primes, and powers}

For $m\in\N$, define $\phi(m)$ to be the number of unit classes in
$\Z/m\Z$.  This defines a function $\phi\colon\N\to\N$, called
\emph{Euler's totient function}.

\examquestion{\label{q:totient}
Let $m,n\in\N$.
\begin{subquestions}
  \item \nameditem{TPP}{Totient of a Prime Power}
  If $p\in\N$ is prime and $r\in\N$, prove
  \[
    \phi(p^r)=p^r-p^{r-1}.
  \]
  \item \nameditem{MTF}{Multiplicativity of the Totient Function}
  Use Question \ref{q:crt-map} to prove that $\phi(mn)=\phi(m)\phi(n)$
  when $\gcd(m,n)=1$.
  \item \nameditem{EPFT}{Euler Product Formula for the Totient}
  Deduce from unique prime factorisation that
  \[
    \phi(m)=m
      \prod_{\substack{p\in\N\text{ prime}\\p\mid m}}
      \left(1-\frac1p\right).
  \]
  \item \nameditem{TFIE}{Totient Formula by Inclusion Exclusion}
  Let $t\in\N_0$, and let $p_i\in\N$ be the distinct prime divisors
  of $m$ for $i\in[t]$; for $m=1$ this is the empty list.  Give a second proof of
  the same formula by applying inclusion--exclusion to their multiples in
  $\{0,1,\ldots,m-1\}$.
  \item \nameditem{TNE}{Totient Numerical Example}
  Compute $\phi(2025)$.
\end{subquestions}}

\examquestion{\label{q:wilson}
Let $p\in\N$ be prime.
\begin{subquestions}
  \item \nameditem{SROP}{Square Roots of One modulo a Prime}
  For $x\in\Z$, show that $x^2\equiv1\pmod p$ if and only if
  $x\equiv1\pmod p$ or $x\equiv-1\pmod p$.
  \item \nameditem{WT}{Wilson Theorem}
  Verify the following congruence directly when $p=2$.  When $p$ is
  odd, pair every non-zero class other than $[1]_p$ and $[-1]_p$ with its
  inverse, and prove Wilson's theorem
  \[
    (p-1)!\equiv-1\pmod p.
  \]
  \item \nameditem{WCCP}{Wilson Converse Characterizes Primes}
  Prove the converse: if $n\in\N$, $n>1$, and
  $(n-1)!\equiv-1\pmod n$, then $n$ is prime.
\end{subquestions}}

For a set $X$, a \emph{permutation of $X$} is a bijection $X\to X$.
If $m\in\N$ and $u,v\in\Z$ represent unit classes $[u]_m,[v]_m$, then
$[uv]_m$ is a unit: the product of inverses of the two factors is an
inverse of their product.
Consequently, a product of units is a unit and can be cancelled from a
congruence by multiplying by its inverse.

\examquestion{\label{q:fermat-euler}
Let $p\in\N$ be prime and let $a\in\Z$.  In part (c), also let
$m\in\N$ and assume $\gcd(a,m)=1$.
\begin{subquestions}
  \item \nameditem{MPNPR}{Multiplication Permutes Non-zero Prime Residues}
  If $p\nmid a$, show that the map
  \[
    \mu_a\colon (\Z/p\Z)\setminus\{[0]_p\}
      \longrightarrow(\Z/p\Z)\setminus\{[0]_p\},
    \qquad \mu_a([x]_p)=[ax]_p\quad(x\in\Z),
  \]
  is a permutation.
  \item \nameditem{FLT}{Fermat Little Theorem}
  Under the hypothesis $p\nmid a$, multiply all the classes in this
  permutation to prove
  \[
    a^{p-1}\equiv1\pmod p,
  \]
  using cancellation by the unit class $[(p-1)!]_p$.  Treat separately the
  case $p\mid a$ and deduce
  $a^p\equiv a\pmod p$ for every $a\in\Z$.
  \item \nameditem{FET}{Fermat Euler Theorem}
  For every unit class $[x]_m$, where $x\in\Z$, set
  $\nu_a([x]_m)=[ax]_m$.  Prove that $\nu_a$ is a permutation of the
  $\phi(m)$ unit classes.  By multiplying these classes and cancelling
  their product, deduce the Fermat--Euler theorem
  \[
    a^{\phi(m)}\equiv1\pmod m.
  \]
\end{subquestions}}

The congruences proved in Question \ref{q:fermat-euler}(b) are the two
standard forms of \emph{Fermat's little theorem}.

\examquestion{\label{q:quadratic-residues}
Let $p\in\N$ be an odd prime.  For $a\in\Z$ with $p\nmid a$, the non-zero
class $[a]_p$ is a \emph{quadratic residue modulo $p$} if there exists
$x\in\Z$ such that $a\equiv x^2\pmod p$.
\begin{subquestions}
  \item \nameditem{ECSP}{Equality Criterion for Squares modulo a Prime}
  For $x,y\in\Z$, prove that $x^2\equiv y^2\pmod p$ if and only if
  $x\equiv\pm y\pmod p$.
  \item \nameditem{NNQR}{Number of Non-zero Quadratic Residues}
  Deduce that there are exactly $(p-1)/2$ non-zero quadratic residues
  modulo $p$.
  \item \nameditem{QRT}{Quadratic Residue Tables}
  List the quadratic residues modulo $7$ and modulo $11$.
\end{subquestions}}

\examquestion{\label{q:minus-one}
Let $p\in\N$ be an odd prime.
\begin{subquestions}
  \item \nameditem{CSRMO}{Constructing Square Roots of Minus One}
  If $p=4k+1$ for some $k\in\N_0$, split $(p-1)!$ into two halves
  and use Wilson's theorem to exhibit an $x\in\Z$ satisfying
  $x^2\equiv-1\pmod p$.
  \item \nameditem{NSRMO}{Non-existence of Square Roots of Minus One}
  If $p=4k+3$ for some $k\in\N_0$, use Fermat's little theorem to
  show that there is no $x\in\Z$ satisfying $x^2\equiv-1\pmod p$.
  \item \nameditem{MOQRC}{Minus One Quadratic Residue Criterion}
  Conclude that $-1$ is a quadratic residue modulo $p$ if and only if
  $p\equiv1\pmod4$.
\end{subquestions}}

\examquestion{\label{q:two-squares}
The aim of this question is to prove the two-squares theorem for primes.
Let $p\in\N$ be a prime satisfying $p\equiv1\pmod4$, and use Question
\ref{q:minus-one} to choose $x\in\Z$ with $x^2\equiv-1\pmod p$.
\begin{subquestions}
  \item \nameditem{ICSP}{Integer between Consecutive Squares below a Prime}
  Use well-ordering to prove that there is a unique $h\in\N_0$ such
  that
  \[
    h^2<p<(h+1)^2.
  \]
  \item \nameditem{CBRP}{Collision among Bounded Residue Pairs}
  Show that the $(h+1)^2$ pairs
  \[
    (a,b)\in
    \{0,1,\ldots,h\}\times\{0,1,\ldots,h\}
  \]
  number more than $p$.  Each pair determines one of the $p$ residue
  classes $[a+bx]_p$; deduce that two distinct pairs determine the same
  class.
  \item \nameditem{SNLRP}{Small Non-zero Linear Relation modulo a Prime}
  By the pigeonhole principle, find $A,B\in\Z$, not both zero,
  such that
  \[
    |A|,|B|\leq h,\qquad A+Bx\equiv0\pmod p.
  \]
  \item \nameditem{PSTS}{Prime as a Sum of Two Squares}
  Prove that $p\mid A^2+B^2$ and
  $0<A^2+B^2<2p$.  Deduce $p=A^2+B^2$.
  \item \nameditem{OPSTSN}{Odd Prime Sum of Two Squares Necessity}
  Conversely, let $q\in\N$ be an odd prime and suppose
  $q=A^2+B^2$ for some $A,B\in\Z$.  Use squares modulo $4$ to prove
  $q\equiv1\pmod4$.
  \item \nameditem{PSTSC}{Prime Sum of Two Squares Criterion}
  Include the prime $2$ and prove the following complete statement:
  a prime $q\in\N$ is a sum of two integer squares if and only if
  $q=2$ or $q\equiv1\pmod4$.
\end{subquestions}
\indication{For part (a), the set
$\{r\in\N:p<r^2\}$ is non-empty; take its least element and use the fact
that a prime is not a perfect square.  For part (c), subtract two distinct
pairs having the same residue.  For part (d), square
$A\equiv-Bx\pmod p$.}}

\examquestion{\label{q:primes-one-mod-four}
Suppose, for a contradiction, that there are only finitely many primes
congruent to $1$ modulo $4$.  Since $5$ is one such prime, choose
$k\in\N$ and list all of them as $p_1,\ldots,p_k\in\N$.  Set
\[
  N=(2p_1\cdots p_k)^2+1\in\N.
\]
\begin{subquestions}
  \item \nameditem{FSCALP}{Fourfold Square Construction Avoids Listed Primes}
  Show that $N$ is odd and that $p_i\nmid N$ for every
  $i\in\{1,\ldots,k\}$.
  \item \nameditem{PDMMOS}{Prime Divisors Make Minus One a Square}
  If $q\in\N$ is any prime divisor of $N$, prove that $q$ is odd
  and that $-1$ is a quadratic residue modulo $q$.
  \item \nameditem{IMPCOF}{Infinitely Many Primes Congruent to One modulo Four}
  Use Question \ref{q:minus-one} to obtain a contradiction.  Deduce
  that there are infinitely many primes congruent to $1$ modulo $4$.
\end{subquestions}}

For $T\in\Z$ and $m\in\N$, the notation $T\bmod m$ means the unique
member $r\in\{0,1,\ldots,m-1\}$ for which $T\equiv r\pmod m$.

\examquestion{\label{q:square-root-formula}
Let $k\in\N_0$, let $p=4k+3\in\N$ be prime, and let $a\in\Z$ satisfy
$p\nmid a$.  Assume that $[a]_p$ is a quadratic residue modulo $p$.
\begin{subquestions}
  \item \nameditem{SRFPTF}{Square Root Formula for Primes Three modulo Four}
  Choose $z\in\Z$ with $a\equiv z^2\pmod p$ and use Fermat's
  theorem to prove
  \[
    (a^{k+1})^2\equiv a\pmod p.
  \]
  \item \nameditem{SRSN}{Square Roots of Seven modulo Nineteen}
  Find both square roots of $7$ modulo $19$.
  \item \nameditem{BERSA}{Binary Expansion and Repeated Squaring Algorithm}
  Let $A\in\Z$, $N\in\N_0$, and $m\in\N$.  When $N=0$, the value
  of $A^N\bmod m$ is immediate.  When $N\geq1$, use repeated division by
  $2$ to write
  \[
    N=\varepsilon_0+\varepsilon_1 2+\cdots+\varepsilon_s2^s,
    \qquad
    \varepsilon_j\in\{0,1\}\ (j\in\{0,1,\ldots,s\}),\quad
    \varepsilon_s=1,
  \]
  for some $s\in\N_0$.  Describe the repeated-squaring algorithm for
  computing $A^N\bmod m$.  Here a \emph{modular multiplication} means
  multiplying two current integer representatives and then taking the
  remainder modulo $m$.  Prove that the algorithm uses at most $2s+1$
  modular multiplications.
  \item \nameditem{MENE}{Modular Exponentiation Numerical Example}
  Compute $3^{2026}\pmod{19}$ by this method.
\end{subquestions}}

\examquestion{\label{q:four-roots}
Let $p,q\in\N$ be distinct odd primes, let $n=pq$, and let $a\in\Z$
satisfy $\gcd(a,n)=1$.  Suppose there exists $u\in\Z$ with
$u^2\equiv a\pmod n$.
\begin{subquestions}
  \item \nameditem{TSRP}{Two Square Roots modulo each Prime}
  Prove that exactly two classes $[x]_p\in\Z/p\Z$, with $x\in\Z$,
  satisfy
  $x^2\equiv a\pmod p$, and that exactly two classes
  $[y]_q\in\Z/q\Z$, with $y\in\Z$, satisfy $y^2\equiv a\pmod q$.
  \item \nameditem{FSRS}{Four Square Roots modulo a Semiprime}
  Combine the two independent sign choices by the Chinese remainder
  theorem and prove that exactly four classes $[z]_n\in\Z/n\Z$, with
  $z\in\Z$, satisfy $z^2\equiv a\pmod n$.
  \item \nameditem{OPFSR}{Opposite Pairing of Four Square Roots}
  For $z,w\in\Z$, call two classes $[z]_n$ and $[w]_n$
  \emph{opposite} if
  $[w]_n=[-z]_n$.  Show that the four roots form two opposite pairs.
\end{subquestions}}

Let $N\in\N$ satisfy $N>1$, and let $a\in\Z$ satisfy $\gcd(a,N)=1$.
A \emph{base-$a$ Fermat check} for $N$ tests whether
$a^{N-1}\equiv1\pmod N$.  Fermat's little theorem shows that every prime
$N$ passes every permitted check, but the converse has not been proved.

\examquestion{\label{q:carmichael}
This question records a limitation of elementary Fermat primality tests.
\begin{subquestions}
  \item \nameditem{FCN}{Factorization of the Carmichael Number}
  Verify $561=3\cdot11\cdot17$.
  \item \nameditem{FCPF}{Fermat Congruence at Prime Factors}
  Let $a\in\Z$ satisfy $\gcd(a,561)=1$.  Use Fermat's little theorem
  modulo $3$, $11$, and $17$ to show that $a^{560}\equiv1$ modulo each of
  these primes.
  \item \nameditem{CCCR}{Carmichael Congruence by Chinese Remainders}
  Apply the Chinese remainder theorem to prove
  $a^{560}\equiv1\pmod{561}$.
  \item \nameditem{FNFPT}{Failure of Naive Fermat Primality Testing}
  Explain why repeatedly checking Fermat's congruence, without further
  safeguards, is not a reliable primality test.
\end{subquestions}}

\exampart{V}{Arithmetic protocols}

This part applies the preceding modular theory to two public protocols.  No
claim about computational hardness is to be proved.

\examquestion{\label{q:factor-from-roots}
Let $p,q\in\N$ be distinct primes, put $n=pq$, and let $u,v\in\Z$
satisfy
\[
  u^2\equiv v^2\pmod n,\qquad
  u\not\equiv v\pmod n,\qquad
  u\not\equiv-v\pmod n.
\]
\begin{subquestions}
  \item \nameditem{NZDSR}{Non-trivial Zero Divisors from Square Roots}
  Prove that $n\mid(u-v)(u+v)$ but that $n$ divides neither factor.
  \item \nameditem{RPFGCD}{Recovering Prime Factors by Greatest Common Divisors}
  Show that $\gcd(n,u-v)$ and $\gcd(n,u+v)$ are the two prime factors
  $p,q$, in some order.
  \item \nameditem{FNSR}{Factorization from Non-opposite Square Roots}
  Deduce that $\gcd(u,n)=\gcd(v,n)=1$.  Explain why finding two
  non-opposite square-root classes of the same unit in $\Z/n\Z$ reveals a
  factorisation of $n$.
\end{subquestions}}

\examquestion{\label{q:coin-protocol}
Alice chooses distinct primes $p,q\in\N$ with
$p\equiv q\equiv3\pmod4$, puts $n=pq$, publishes $n$, and keeps $p$ and
$q$ secret.  Bob chooses $u\in\Z$ with $\gcd(u,n)=1$, sends the class
$[a]_n=[u^2]_n$, where $a\in\{0,1,\ldots,n-1\}$, and remembers the
opposite pair $\{[u]_n,[-u]_n\}$.  Alice computes the four square-root
classes of $[a]_n$ and announces one of their two opposite pairs, choosing
each pair with probability $1/2$.
\begin{subquestions}
  \item \nameditem{CFSRF}{Computing Four Square Roots from a Factorization}
  Give an explicit finite procedure by which Alice can compute the
  roots from $p$ and $q$, using Questions \ref{q:square-root-formula} and
  \ref{q:crt}.
  \item \nameditem{HCPF}{Honest Coin Protocol is Fair}
  Show that Bob wins with probability $1/2$ if he declares a win
  exactly when Alice does not announce
  $\{[u]_n,[-u]_n\}$.
  \item \nameditem{CSRF}{Cheating Strategy Reveals a Factor}
  If Bob can name the other pair after Alice announces
  $\{[u]_n,[-u]_n\}$, show how Bob can factor $n$.
  \item \nameditem{CAMC}{Computational Assumption versus Mathematical Conclusion}
  State carefully what mathematical conclusion this gives about
  cheating, without treating the presumed difficulty of factorisation as a
  theorem.
\end{subquestions}}

\examquestion{\label{q:rsa-keys}
Let $p,q\in\N$ be distinct primes and let $e\in\N$ satisfy
$1<e<(p-1)(q-1)$ and
\[
  n=pq\in\N,\qquad \varphi=\phi(n)=(p-1)(q-1)\in\N,
  \qquad \gcd(e,\varphi)=1.
\]
\begin{subquestions}
  \item \nameditem{EUPE}{Existence and Uniqueness of the Private Exponent}
  Prove that there is a unique class $[d]_\varphi$, with $d\in\Z$,
  such that
  $ed\equiv1\pmod\varphi$.
  \item \nameditem{CPEEE}{Computing the Private Exponent by Extended Euclid}
  Explain how the extended Euclidean algorithm computes the least
  positive representative $d\in\{1,2,\ldots,\varphi-1\}$ of this class.
  \item \nameditem{EPD}{Exponent Product Decomposition}
  Write $ed=1+k\varphi$ for some $k\in\N_0$.
\end{subquestions}
The pair $(n,e)$ is the public key and $d$ is the private exponent.  A
\emph{message} is an integer $M\in\{0,1,\ldots,n-1\}$.  Its
\emph{ciphertext} is the unique integer
$C\in\{0,1,\ldots,n-1\}$ such that $C\equiv M^e\pmod n$.}

\examquestion{\label{q:rsa-coprime}
Retain the notation and hypotheses of Question \ref{q:rsa-keys}.  Let
$M,C\in\{0,1,\ldots,n-1\}$ be a message and its ciphertext, and assume
$\gcd(M,n)=1$.
\begin{subquestions}
  \item \nameditem{ECCM}{Euler Congruence for Coprime Messages}
  Apply the Fermat--Euler theorem to $M^\varphi$.
  \item \nameditem{RSACMC}{Rivest Shamir Adleman Coprime Message Correctness}
  Deduce that
  \[
    C^d\equiv M^{ed}\equiv M\pmod n.
  \]
\end{subquestions}}

\examquestion{\label{q:rsa-all-messages}
Retain $p,q,n,e,d,\varphi$, and $k$ from Question \ref{q:rsa-keys}, and
let $M\in\Z$ be arbitrary.  No coprimality assumption is made.
\begin{subquestions}
  \item \nameditem{DCFP}{Decryption Correctness modulo the First Prime}
  Prove $M^{ed}\equiv M\pmod p$, separating the cases $p\mid M$ and
  $p\nmid M$.
  \item \nameditem{DCSP}{Decryption Correctness modulo the Second Prime}
  Prove the analogous congruence modulo $q$.
  \item \nameditem{RSACEM}{Rivest Shamir Adleman Correctness for Every Message}
  Use the Chinese remainder theorem to conclude
  $M^{ed}\equiv M\pmod n$ for every integer $M$.
\end{subquestions}
\indication{Since $(p-1)\mid\varphi$, the relation
$ed=1+k\varphi$, where $k\in\N_0$, may be used with Fermat's little
theorem modulo $p$.}}

\examquestion{\label{q:rsa-example}
\begin{subquestions}
  \item \nameditem{RSAKGE}{Rivest Shamir Adleman Key Generation Example}
  Take $p=5$, $q=11$, and $e=3$.  Compute $n$, $\varphi$, and the
  least positive private exponent $d$.
  \item \nameditem{RSAEDE}{Rivest Shamir Adleman Encryption and Decryption Example}

  \item \nameditem{TRPS}{Totient Reveals the Prime Sum}

\exampart{VI}{Constructing and completing the number line}

This part constructs $\Z$ and $\Q$ as quotient sets and then studies the
completeness property which distinguishes $\R$ from $\Q$.

\subpart{A. --- From successors to fractions}

For a set $X$, a \emph{binary operation on $X$} is a map
$X\times X\to X$.

\examquestion{\label{q:peano-arithmetic}
Regard $\N_0$ abstractly as a set with a distinguished element $0$ and an
injective successor map $S\colon\N_0\to\N_0$ whose image does not contain
$0$.  Assume the induction axiom from Question \ref{q:peano-induction}.
Put $1=S(0)$.  The recursion theorem admitted after Question
\ref{q:well-ordering} gives unique binary operations $+$ and $\cdot$ on
$\N_0$ satisfying, for every $n,m\in\N_0$,
\[
  n+0=n,\qquad n+S(m)=S(n+m),
\]
\[
  n\cdot0=0,\qquad n\cdot S(m)=n\cdot m+n.
\]
\begin{subquestions}
  \item \nameditem{SRIA}{Successor Recursion Identities for Addition}
  For every $n,m\in\N_0$, prove
  $0+n=n$ and $S(n)+m=S(n+m)$.
  \item \nameditem{CPA}{Commutativity of Peano Addition}
  For every $n,m\in\N_0$, deduce $n+m=m+n$.
  \item \nameditem{APA}{Associativity of Peano Addition}
  For every $n,m,r\in\N_0$, prove
  $(n+m)+r=n+(m+r)$.
  \item \nameditem{PMIC}{Peano Multiplication Identities and Commutativity}
  For every $n,m\in\N_0$, prove
  $0\cdot n=0$ and $S(n)\cdot m=n\cdot m+m$, and deduce
  $n\cdot m=m\cdot n$.
  \item \nameditem{DLPA}{Distributive Laws for Peano Arithmetic}
  For every $n,m,r\in\N_0$, prove
  $n\cdot(m+r)=n\cdot m+n\cdot r$ and
  $(n+m)\cdot r=n\cdot r+m\cdot r$.
  \item \nameditem{AMAC}{Associativity of Multiplication and Additive Cancellation}
  For every $n,m,r\in\N_0$, prove
  $(n\cdot m)\cdot r=n\cdot(m\cdot r)$ and prove additive cancellation:
  $n+r=m+r$ implies $n=m$.
  \item \nameditem{TOPA}{Total Order from Peano Addition}
  Define $n\leq m$ to mean that $m=n+r$ for some $r\in\N_0$.
  Prove that $\leq$ is a total order on $\N_0$.  Deduce that, for every
  $n,m\in\N_0$, exactly one of the following holds:
  \[
    n=m,\qquad n=m+S(r)\text{ for some }r\in\N_0,\qquad
    m=n+S(r)\text{ for some }r\in\N_0.
  \]
  \item \nameditem{ZPPNN}{Zero Product Property for Natural Numbers}
  For every $n,m\in\N_0$, prove that $n\cdot m=0$ implies $n=0$ or
  $m=0$.
\end{subquestions}
\indication{Use induction on the variable occurring in the second position
of a recursively defined operation.  Establish each displayed auxiliary
identity before using it in the next part.  For part (g), first prove by
induction that every element other than $0$ lies in the image of $S$.  For
part (h), induct on $m$.}}

For $n,m\in\N_0$, write $n<m$ when $n\leq m$ and $n\ne m$.
A relation $<$ on a set $X$ is a \emph{strict total order} if, for all
$x,y,z\in X$, it has the following properties:
\[
  x\nless x;\qquad
  x<y\text{ and }y<z\Longrightarrow x<z;
\]
and exactly one of $x<y$, $x=y$, and $y<x$ holds.

A \emph{commutative ring} is a set $R$ with binary operations $+$ and
$\cdot$, distinguished elements $0,1\in R$, and, for each $x\in R$, an
element $-x\in R$.  For all $x,y,z\in R$, the following identities are
required:
\[
  (x+y)+z=x+(y+z),\quad x+y=y+x,\quad x+0=x,\quad x+(-x)=0,
\]
\[
  (xy)z=x(yz),\quad xy=yx,\quad x1=x,
\]
\[
  x(y+z)=xy+xz,\qquad (x+y)z=xz+yz.
\]

\examquestion{\label{q:integers}
For $(a,b),(c,d)\in\N_0^2$, define
\[
  (a,b)\sim(c,d)\quad\Longleftrightarrow\quad a+d=b+c.
\]
We define the set of integers to be the quotient
$\Z=\N_0^2/{\sim}$ and write $[a,b]_{\Z}$ for the class of
$(a,b)\in\N_0^2$.
\begin{subquestions}
  \item \nameditem{IPRER}{Integer-Pair Relation is an Equivalence Relation}
  Prove that $\sim$ is an equivalence relation.
  \item \nameditem{WDAIC}{Well-Defined Addition of Integer Classes}
  Prove that
  \[
    [a,b]_{\Z}+[c,d]_{\Z}=[a+c,b+d]_{\Z}
  \]
  is independent of the representatives.
  \item \nameditem{WDMIC}{Well-Defined Multiplication of Integer Classes}
  Prove the same for
  \[
    [a,b]_{\Z}[c,d]_{\Z}=[ac+bd,ad+bc]_{\Z}.
  \]
  \item \nameditem{ICIAI}{Integer Class Identities and Additive Inverses}
  For $a,b\in\N_0$, identify the additive and multiplicative
  identities and the additive inverse of $[a,b]_{\Z}$.
  \item \nameditem{NNEI}{Natural Number Embedding into the Integers}
  Define
  \[
    \iota\colon\N_0\longrightarrow\Z,
    \qquad \iota(n)=[n,0]_{\Z}\quad(n\in\N_0).
  \]
  Prove that $\iota$ is injective and that, for all $n,m\in\N_0$,
  \[
    \iota(0)=0,\quad \iota(1)=1,\quad
    \iota(n+m)=\iota(n)+\iota(m),\quad
    \iota(nm)=\iota(n)\iota(m).
  \]
  Using Question \ref{q:peano-arithmetic}(g), prove that
  every class has the form $[n,0]_{\Z}$ or $[0,n]_{\Z}$ for some
  $n\in\N_0$.
  \item \nameditem{ICSCRL}{Integer Classes Satisfy Commutative Ring Laws}
  Using the identities proved in Question \ref{q:peano-arithmetic},
  verify associativity, commutativity, and distributivity for these
  operations.  Conclude that the quotient is a commutative ring.
  \item \nameditem{STOIC}{Strict Total Order on Integer Classes}
  Define
  \[
    [a,b]_{\Z}<[c,d]_{\Z}\quad\Longleftrightarrow\quad a+d<b+c.
  \]
  Prove that this is independent of representatives and gives a strict
  total order.
  \item \nameditem{OCIO}{Order Compatibility with Integer Operations}
  For all $\alpha,\beta,\gamma\in\Z$, prove that
  $\alpha<\beta$ implies $\alpha+\gamma<\beta+\gamma$, and that
  $\alpha<\beta$ and $0<\gamma$ imply
  $\alpha\gamma<\beta\gamma$.
  \item \nameditem{ZPCI}{Zero Product and Cancellation in the Integers}
  For $\alpha,\beta\in\Z$, use the two forms in part (e) and
  Question \ref{q:peano-arithmetic}(h) to prove that
  $\alpha\beta=0$ implies $\alpha=0$ or $\beta=0$.  Deduce that, for
  $\alpha,\beta,\gamma\in\Z$ with $\gamma\ne0$, the equality
  $\alpha\gamma=\beta\gamma$ implies $\alpha=\beta$.
\end{subquestions}}
\indication{For part (h), first prove from the definition of the order on
$\N_0$ that, if $a,b\in\N_0$ satisfy $a<b$ and $t\in\N$, then
$at<bt$.  Use part (e) to represent a positive integer class by
$[t,0]_{\Z}$ with $t\in\N$.}

From now on, identify each $n\in\N_0$ with its image
$\iota(n)\in\Z$ from Question \ref{q:integers}(e).  In particular,
$\N$ is identified with the set of positive elements
$\{\iota(n):n\in\N\}$ of $\Z$.

A \emph{field} is a commutative ring $R$ in which $0\ne1$ and every
non-zero element $x\in R$ has a multiplicative inverse: there exists
$y\in R$ such that $xy=1$.

\examquestion{\label{q:rationals}
For $(a,b),(c,d)\in\Z\times\N$, define
\[
  (a,b)\approx(c,d)\quad\Longleftrightarrow\quad ad=bc,
\]
and define the set of rational numbers to be
$\Q=(\Z\times\N)/{\approx}$.  Write $a/b$ for the class of
$(a,b)\in\Z\times\N$.
\begin{subquestions}
  \item \nameditem{RPRER}{Rational-Pair Relation is an Equivalence Relation}
  Prove that $\approx$ is an equivalence relation.
  \item \nameditem{WDRA}{Well-Defined Rational Arithmetic}
  Prove that the rules
  \[
    \frac ab+\frac cd=\frac{ad+bc}{bd},
    \qquad
    \frac ab\cdot\frac cd=\frac{ac}{bd}
  \]
  are independent of representatives.
  \item \nameditem{RII}{Rational Identities and Inverses}
  For $a\in\Z$ and $b\in\N$, find the additive inverse of $a/b$.
  Prove that $0/1$ and $1/1$ are respectively the additive and
  multiplicative identities and that $0/1\ne1/1$.
  When $a\ne0$, construct the multiplicative inverse of $a/b$ with a
  positive denominator, treating separately the cases $a>0$ and $a<0$.
  \item \nameditem{RCFF}{Rational Classes Form a Field}
  Verify associativity, commutativity, the identity laws, and
  distributivity
  for the two operations.  Together with part (c), conclude that these
  classes form a field.  You may cite the corresponding integer laws.
  \item \nameditem{STORC}{Strict Total Order on Rational Classes}
  For $a,c\in\Z$ and $b,d\in\N$, show that $a/b<c/d$ if and only if
  $ad<bc$ defines a strict total order independently of representatives.
  \item \nameditem{IER}{Integer Embedding into the Rationals}
  Define $\kappa\colon\Z\to\Q$ by
  $\kappa(a)=a/1$ for every $a\in\Z$.  Prove that $\kappa$ is injective
  and preserves $0$, $1$, addition, multiplication, and order.  From now
  on, identify $a\in\Z$ with $\kappa(a)\in\Q$.
\end{subquestions}}
\indication{For transitivity of $\approx$ and for independence of
representatives, use multiplicative cancellation in $\Z$ from Question
\ref{q:integers}(i).}

If $<$ is a strict total order on a set $X$ and $x,y\in X$, write
$x\leq y$ when $x<y$ or $x=y$.  A \emph{totally ordered field} is a
field $F$ with a total order $\leq$ such that, for all $x,y,z\in F$,
\[
  x\leq y\Longrightarrow x+z\leq y+z,
  \qquad
  x\leq y\text{ and }0\leq z\Longrightarrow xz\leq yz.
\]
An ordered set $X$ is \emph{densely ordered} if, whenever $x,y\in X$ and
$x<y$, there exists $z\in X$ such that $x<z<y$.

\examquestion{\label{q:ordered-rationals}
\begin{subquestions}
  \item \nameditem{ROTI}{Rational Order is Translation-Invariant}
  For $r,s,t\in\Q$ with $r<s$, prove $r+t<s+t$.
  \item \nameditem{RORPM}{Rational Order Respects Positive Multiplication}
  For $r,s,t\in\Q$ with $r<s$ and $t>0$, prove $rt<st$.
  \item \nameditem{RFTOF}{Rationals Form a Totally Ordered Field}
  Combine the field laws from Question \ref{q:rationals} with parts
  (a) and (b) to conclude that $\Q$ is a totally ordered field.
  \item \nameditem{RMDO}{Rational Midpoints and Dense Order}
  If $r,s\in\Q$ satisfy $r<s$, prove
  \[
    r<\frac{r+s}{2}<s.
  \]
  Thus $\Q$ is densely ordered in itself.
\end{subquestions}}

\examquestion{\label{q:sqrt-two-irrational}
Suppose, for a contradiction, that $x\in\Q$ satisfies $x^2=2$.
\begin{subquestions}
  \item \nameditem{LTCRSR}{Lowest Terms Consequence of a Rational Square Root}
  By the definition of $\Q$, write $x=a/b$ with $a\in\Z$ and
  $b\in\N$.  Prove that $a\ne0$.  Put
  $d=\gcd(a,b)\in\N$, $p=a/d\in\Z$, and
  $q=b/d\in\N$.  Prove that $x=p/q$, that $\gcd(p,q)=1$, and that
  $p^2=2q^2$.
  \item \nameditem{ENES}{Even Numerator from an Even Square}
  Use Question \ref{q:proofs} to prove that $p$ is even.
  \item \nameditem{ISRT}{Irrationality of the Square Root of Two}
  Write $p=2r$ for some $r\in\Z$ and prove that $q$ is even.
  Obtain a contradiction and conclude that there is no
  $x\in\Q$ satisfying $x^2=2$.
\end{subquestions}}

Let $X$ be a totally ordered set and let $A\subseteq X$.  An element
$u\in X$ is an \emph{upper bound} for $A$ if $a\leq u$ for every
$a\in A$.  It is a \emph{least upper bound}, or \emph{supremum}, if it is
an upper bound and $u\leq v$ for every upper bound $v\in X$ of $A$; when
it exists it is denoted $\sup A$.  An element $\ell\in X$ is a
\emph{lower bound} for $A$ if $\ell\leq a$ for every $a\in A$.  It is a
\emph{greatest lower bound}, or \emph{infimum}, if it is a lower bound and
$v\leq\ell$ for every lower bound $v\in X$ of $A$; when it exists it is
denoted $\inf A$.

\examquestion{\label{q:q-incomplete}
Let
\[
  E=\{x\in\Q:x\geq0,\ x^2<2\}.
\]
\begin{subquestions}
  \item \nameditem{RSBSNB}{Rational Square-Bound Set is Non-empty and Bounded}
  Show that $E$ is non-empty and bounded above in $\Q$.
  \item \nameditem{PRSA}{Positive Rational Supremum Assumption}
  Suppose, for a contradiction, that there exists $s\in\Q$ which is
  the supremum of $E$ in the ordered set $\Q$.  Prove that $s>0$.
  \item \nameditem{UPPRS}{Upward Perturbation of a Putative Rational Supremum}
  Under the supposition of part (b), if $s^2<2$, choose
  $h\in\Q$ with $h>0$ such that
  \[
    h<1,\qquad h<\frac{2-s^2}{2s+1},
  \]
  and prove $(s+h)^2<2$.  Obtain a contradiction.
  \item \nameditem{DPPRS}{Downward Perturbation of a Putative Rational Supremum}
  Under the supposition of part (b), if $s^2>2$, choose
  $h\in\Q$ with $h>0$ such that
  \[
    h<s,\qquad h<\frac{s^2-2}{2s},
  \]
  and prove that $s-h$ is an upper bound for $E$.  Obtain a contradiction.
  \item \nameditem{ROC}{Rational Order is not Complete}
  Use Question \ref{q:sqrt-two-irrational} to prove that one of
  parts (c) and (d) applies, and conclude that $E$ has no supremum in
  $\Q$.
\end{subquestions}}

\subpart{B. --- Completeness}

\concept{The real numbers and intervals}
From now on, $\R$ is a totally ordered field with a fixed injective map
$\jmath\colon\Q\to\R$.  For all $q,r\in\Q$, it satisfies
\[
  \jmath(0)=0,\quad \jmath(1)=1,\quad
  \jmath(q+r)=\jmath(q)+\jmath(r),\quad
  \jmath(qr)=\jmath(q)\jmath(r),
\]
and $q<r$ if and only if $\jmath(q)<\jmath(r)$.  We identify each
$q\in\Q$ with
$\jmath(q)\in\R$ and in this sense say that $\R$ \emph{contains} $\Q$.
The field $\R$ satisfies the \emph{least-upper-bound axiom}: every
non-empty subset of $\R$ which is bounded above has a supremum in $\R$.

For $a,b\in\R$ with $a\leq b$, define the intervals
\[
  [a,b]=\{x\in\R:a\leq x\leq b\},\qquad
  (a,b)=\{x\in\R:a<x<b\},
\]
\[
  [a,b)=\{x\in\R:a\leq x<b\},\qquad
  (a,b]=\{x\in\R:a<x\leq b\}.
\]

\examquestion{\label{q:sup-inf}
\begin{subquestions}
  \item \nameditem{ULUB}{Uniqueness of Least Upper Bounds}
  Let $X$ be a totally ordered set and let $A\subseteq X$.  Prove
  that a least upper bound of $A$ in $X$, when it exists, is unique.
  \item \nameditem{GLBLUB}{Greatest Lower Bounds from Least Upper Bounds}
  Let $A\subseteq\R$ be non-empty and bounded below.  Apply the
  least-upper-bound axiom to $-A=\{-a:a\in A\}$ and prove
  \[
    \inf A=-\sup(-A).
  \]
  \item \nameditem{SIBS}{Suprema and Infima of Basic Sets}
  Find the supremum and infimum of each of
  $(0,1)$, $[0,1)$, and $\{1/n:n\in\N\}$, postponing only the proof of the
  last infimum to Question \ref{q:archimedean-density}.
\end{subquestions}}

\examquestion{\label{q:archimedean}
Suppose, for a contradiction, that $\N$ is bounded above in $\R$, and let
$s=\sup\N$.
\begin{subquestions}
  \item \nameditem{PPNUB}{Predecessor of a Putative Natural Upper Bound}
  Show that $s-1$ is not an upper bound for $\N$.
  \item \nameditem{CNUB}{Contradiction to a Natural Upper Bound}
  Find $n\in\N$ with $n>s-1$ and obtain a contradiction from $n+1$.
  \item \nameditem{APR}{Archimedean Property of the Reals}
  Conclude that for every $x\in\R$ there exists $n\in\N$ with $n>x$.
\end{subquestions}}

The conclusion of Question \ref{q:archimedean} is called the
\emph{Archimedean property of $\R$}.

\examquestion{\label{q:archimedean-density}
\begin{subquestions}
  \item \nameditem{RNIZ}{Reciprocal Naturals have Infimum Zero}
  Use the Archimedean property to prove
  \[
    \inf\{1/n:n\in\N\}=0.
  \]
  \item \nameditem{SRRG}{Small Reciprocal below a Real Gap}
  Let $r,s\in\R$ satisfy $r<s$.  Choose $n\in\N$ with
  $1/n<s-r$.
  \item \nameditem{LIRM}{Least Integer above a Real Multiple}
  Let $m\in\Z$ be the least integer satisfying $m>nr$, and prove
  $m-1\leq nr<m$.
  \item \nameditem{DRR}{Density of the Rationals in the Reals}
  Deduce
  \[
    r<\frac mn<s,
  \]
  and hence prove that $\Q$ is dense in $\R$.
\end{subquestions}
\indication{For part (c), first choose $k\in\N$ with $k>-nr$, and then
choose $H\in\N$ with $H>nr+k$.  The set
$\{j\in\N_0:j>nr+k\}$ is non-empty.  If $j_0$ is its least element, prove
that $m=j_0-k$ is the least integer greater than $nr$.}}

\examquestion{\label{q:real-sqrt-two}
In $\R$, let
\[
  F=\{x\in\R:x\geq0,\ x^2<2\}.
\]
\begin{subquestions}
  \item \nameditem{RSBCSNB}{Real Square-Bound Candidate Set is Non-empty and Bounded}
  Show that $F$ is non-empty and bounded above.
  \item \nameditem{RSSBT}{Real Supremum cannot Square Below Two}
  Let $s=\sup F\in\R$.  Prove that $s>0$, and adapt Question
  \ref{q:q-incomplete}(c) to rule out $s^2<2$.
  \item \nameditem{RSSAT}{Real Supremum cannot Square Above Two}
  Adapt Question \ref{q:q-incomplete}(d) to rule out $s^2>2$.
  \item \nameditem{ERSRT}{Existence of the Real Square Root of Two}
  Conclude that $s^2=2$ and hence that $\sqrt2$ exists in $\R$.
\end{subquestions}
\indication{The two perturbations displayed in Question
\ref{q:q-incomplete} are real here; only their positivity and size matter.}}

\examquestion{\label{q:dedekind}
A \emph{Dedekind lower cut} is a set $L\subset\Q$ satisfying all four
conditions below:
\begin{properties}
  \item $L\ne\varnothing$;
  \item $L\ne\Q$;
  \item if $q\in L$ and $r\in\Q$ satisfies $r<q$, then $r\in L$;
  \item for every $q\in L$, there exists $r\in L$ with $q<r$.
\end{properties}
Cuts are ordered by inclusion.
\begin{subquestions}
  \item \nameditem{ERNDDC}{Every Real Number Defines a Dedekind Cut}
  For $x\in\R$, prove that
  \[
    L_x=\{q\in\Q:q<x\}
  \]
  is a cut.
  \item \nameditem{RNEDC}{Real Numbers Embed into Dedekind Cuts}
  Use the density of $\Q$ in $\R$ to prove that $x\mapsto L_x$ is
  injective and order-preserving.
  \item \nameditem{UUBFC}{Union of an Upper-Bounded Family of Cuts}
  Let $\mathcal C$ be a non-empty set of cuts.  Suppose that there
  is a cut $U$ such that $L\subseteq U$ for every $L\in\mathcal C$.
  Prove that $\bigcup_{L\in\mathcal C}L$ is a cut.
  \item \nameditem{CODC}{Completeness of the Ordered Dedekind Cuts}
  Prove that this union is the least upper bound of $\mathcal C$.
  Explain how this establishes completeness of the ordered set of cuts.
\end{subquestions}}

\admitted{DCFC}{Dedekind Cut Field Construction}{Let $\mathcal D$ be the set
of all Dedekind lower cuts.  There exist binary operations
$\oplus,\odot\colon\mathcal D\times\mathcal D\to\mathcal D$
and elements $0_{\mathcal D},1_{\mathcal D}\in\mathcal D$ for which
$\mathcal D$, ordered by inclusion, is a totally ordered field satisfying the
least-upper-bound axiom.  Moreover, the map
$j\colon\Q\to\mathcal D$ defined by
$j(q)=\{r\in\Q:r<q\}$ for every $q\in\Q$ is injective and, for all
$q,r\in\Q$, satisfies
\[
  j(0)=0_{\mathcal D},\qquad j(1)=1_{\mathcal D},\qquad
  j(q+r)=j(q)\oplus j(r),\qquad j(qr)=j(q)\odot j(r),
\]
and $q\leq r$ if and only if $j(q)\subseteq j(r)$.  Thus $\mathcal D$ gives
a construction of $\R$.}

\exampart{VII}{Limits, digits, and exceptional numbers}

This part develops elementary convergence from completeness, then uses
series and decimal representations to exhibit irrational and transcendental
numbers.

\subpart{A. --- Sequences and series}

For $x\in\R$, define its \emph{absolute value} by
\[
  |x|=\begin{cases}x,&x\geq0,\\-x,&x<0.\end{cases}
\]
A \emph{real sequence} is a map $a\colon\N\to\R$.  For each
$n\in\N$, its $n$th term is denoted by $a_n=a(n)$, and the sequence itself
is denoted by $(a_n)_{n\in\N}$.  Let $\ell\in\R$.  We say that
$(a_n)_{n\in\N}$ \emph{tends to} $\ell$ if, for every
$\varepsilon\in\R$ with $\varepsilon>0$, there exists $N\in\N$ such that
\[
  |a_n-\ell|<\varepsilon
  \quad\hbox{for every $n\in\N$ with $n\geq N$.}
\]
In this case we write $a_n\to\ell$ as $n\to\infty$, or
$\lim_{n\to\infty}a_n=\ell$, and call $\ell$ a \emph{limit} of the
sequence.  A sequence \emph{converges} if it tends to some real number; it
\emph{diverges} if it does not converge.

\examquestion{\label{q:sequence-definition}
\begin{subquestions}
  \item \nameditem{AVLTI}{Absolute Value Laws and Triangle Inequality}
  For all $x,y\in\R$, prove directly from the definition of absolute
  value that
  \[
    |x|\geq0,\qquad -|x|\leq x\leq|x|,\qquad
    |xy|=|x||y|,\qquad |x+y|\leq|x|+|y|.
  \]
  The last inequality is called the \emph{triangle inequality}.
  \item \nameditem{NSLD}{Negation of the Sequence Limit Definition}
  Write the exact negation of the displayed definition of
  $a_n\to\ell$.
  \item \nameditem{RSCZ}{Reciprocal Sequence Converges to Zero}
  For every $n\in\N$, set $a_n=1/n$.  Use the Archimedean property
  to prove that $a_n\to0$.
  \item \nameditem{ASSD}{Alternating Sign Sequence Diverges}
  For every $n\in\N$, set $b_n=(-1)^n$.  Prove directly from the
  negated definition that $(b_n)_{n\in\N}$ diverges.
  \item \nameditem{CSC}{Constant Sequence Convergence}
  Let $c\in\R$, and let $c_n=c$ for every $n\in\N$.  Prove that
  $c_n\to c$.
  \item \nameditem{EESHSL}{Eventually Equal Sequences Have the Same Limit}
  Let $(u_n)_{n\in\N}$ and $(v_n)_{n\in\N}$ be real sequences.
  Suppose that there exists $N_0\in\N$ such that $u_n=v_n$ for every
  $n\in\N$ with $n\geq N_0$.  Prove that one sequence converges if and only
  if the other does, and that their limits then agree.
\end{subquestions}}

A real sequence $(a_n)_{n\in\N}$ is \emph{bounded} if there exists
$B\in\R$ with $B\geq0$ such that $|a_n|\leq B$ for every $n\in\N$.
Let $n\colon\N\to\N$ be a function, write $n_j=n(j)$ for every
$j\in\N$, and suppose that $n_j<n_{j+1}$ for every $j\in\N$.  Then
$(a_{n_j})_{j\in\N}$ is called a \emph{subsequence} of
$(a_n)_{n\in\N}$.

\examquestion{\label{q:limit-basic}
Let $(a_n)_{n\in\N}$ be a real sequence and let $a\in\R$.  Suppose that
$a_n\to a$.
\begin{subquestions}
  \item \nameditem{CSB}{Convergent Sequences are Bounded}
  Prove that $(a_n)$ is bounded.
  \item \nameditem{USL}{Uniqueness of Sequence Limits}
  Let $b\in\R$.  If also $a_n\to b$, use
  $|a-b|\leq|a-a_n|+|a_n-b|$ to prove $a=b$.
  \item \nameditem{SPL}{Subsequences Preserve Limits}
  Let $n\colon\N\to\N$ satisfy $n_j<n_{j+1}$ for every
  $j\in\N$.  Prove that the subsequence $(a_{n_j})_{j\in\N}$ also
  converges to $a$.
\end{subquestions}}

\examquestion{\label{q:limit-algebra}
Let $(a_n)_{n\in\N}$ and $(b_n)_{n\in\N}$ be real sequences, and let
$a,b\in\R$.  Suppose that $a_n\to a$ and $b_n\to b$.
\begin{subquestions}
  \item \nameditem{SLSL}{Sum Law for Sequence Limits}
  Prove $a_n+b_n\to a+b$.
  \item \nameditem{PLSL}{Product Law for Sequence Limits}
  Use boundedness and
  \[
    a_nb_n-ab=a_n(b_n-b)+b(a_n-a)
  \]
  to prove $a_nb_n\to ab$.
  \item \nameditem{RLSL}{Reciprocal Law for Sequence Limits}
  If $a\ne0$ and $a_n\ne0$ for every $n\in\N$, prove
  $1/a_n\to1/a$.
  \item \nameditem{STS}{Squeeze Theorem for Sequences}
  Let $(u_n)_{n\in\N}$, $(v_n)_{n\in\N}$, and
  $(c_n)_{n\in\N}$ be real sequences, and let $\ell\in\R$.
  Suppose that $u_n\to\ell$, $v_n\to\ell$, and
  $u_n\leq c_n\leq v_n$ for every $n\in\N$.  Prove $c_n\to\ell$.
\end{subquestions}
\indication{For part (c), first show that
$|a_n|\geq|a|/2$ for every sufficiently large $n$.}}

A real sequence $(a_n)_{n\in\N}$ is \emph{increasing} if
$a_m\leq a_n$ whenever $m,n\in\N$ and $m\leq n$.  It is
\emph{decreasing} if $a_m\geq a_n$ whenever $m,n\in\N$ and $m\leq n$.
It is \emph{strictly increasing} if $a_m<a_n$ whenever
$m,n\in\N$ and $m<n$.
It is \emph{monotone} if it is increasing or decreasing.  It is
\emph{bounded above} if there exists $M\in\R$ such that $a_n\leq M$ for
every $n\in\N$.

\examquestion{\label{q:monotone-convergence}
Let $(a_n)_{n\in\N}$ be an increasing real sequence which is bounded above,
and put
$s=\sup\{a_n:n\in\N\}$.
\begin{subquestions}
  \item \nameditem{SAST}{Supremum Approximation by Sequence Terms}
  Given $\varepsilon\in\R$ with $\varepsilon>0$, explain why
  $s-\varepsilon$ is not an upper bound for the set of terms.
  \item \nameditem{IBSCS}{Increasing Bounded Sequence Converges to its Supremum}
  Choose $N\in\N$ with $a_N>s-\varepsilon$ and prove that
  $|a_n-s|<\varepsilon$ for every $n\in\N$ with $n\geq N$.
  \item \nameditem{MCTS}{Monotone Convergence Theorem for Sequences}
  Let $(b_n)_{n\in\N}$ be a decreasing bounded real sequence.
  Apply the preceding result to $(-b_n)_{n\in\N}$ and prove that
  $(b_n)_{n\in\N}$ converges.  State the resulting monotone convergence
  theorem.
\end{subquestions}}

\examquestion{\label{q:monotone-subsequence}
Let $(a_n)_{n\in\N}$ be a real sequence.  Call $n\in\N$ a \emph{peak}
of this sequence if $a_n\geq a_m$ for every $m\in\N$ with $m>n$.
\begin{subquestions}
  \item \nameditem{IMPDS}{Infinitely Many Peaks give a Decreasing Subsequence}
  If there are infinitely many peaks, construct a decreasing
  subsequence.
  \item \nameditem{FMPIS}{Finitely Many Peaks give an Increasing Subsequence}
  If there are only finitely many peaks, construct a strictly
  increasing subsequence.
  \item \nameditem{MST}{Monotone Subsequence Theorem}
  Conclude that every real sequence has a monotone subsequence.
  \item \nameditem{BWTS}{Bolzano Weierstrass Theorem for Sequences}
  Deduce that every bounded real sequence has a convergent subsequence.
\end{subquestions}
\indication{After the last peak, every term has a strictly larger term
somewhere to its right.}}

Let $(u_n)_{n\in\N}$ be a real sequence.  For each $N\in\N$, the number
\[
  S_N=\sum_{n=1}^{N}u_n
\]
is its $N$th \emph{partial sum}.  If $(S_N)_{N\in\N}$ converges to a real
number $S$, we say that the \emph{series} with terms $u_n$ converges to
$S$, and write $\sum_{n=1}^{\infty}u_n=S$.  Otherwise the series diverges.

\examquestion{\label{q:series}
\begin{subquestions}
  \item \nameditem{TSE}{Telescoping Series Evaluation}
  For every $N\in\N$ with $N\geq2$, show that
  \[
    \sum_{n=2}^N\frac1{n(n-1)}=1-\frac1N
  \]
  and evaluate the corresponding infinite series.
  \item \nameditem{CTRS}{Comparison Test for Reciprocal Squares}
  Compare $1/n^2$ with $1/(n(n-1))$ for every $n\in\N$ with
  $n\geq2$, and prove that
  $\sum_{n=1}^{\infty}1/n^2$ converges.
  \item \nameditem{IGSF}{Infinite Geometric Series Formula}
  Let $r\in\R$ satisfy $|r|<1$.  For $N\in\N$, compute the finite
  sum $\sum_{n=1}^N r^n$ and deduce the value of the infinite geometric
  series.
  \item \nameditem{DHS}{Divergence of the Harmonic Series}
  The \emph{harmonic series} is $\sum_{n=1}^{\infty}1/n$.  For each
  $j\in\N$, group together the terms whose indices $n\in\N$ satisfy
  $2^{j-1}<n\leq2^j$, and prove that the harmonic series diverges.
\end{subquestions}}
\indication{For part (b), the partial sums are increasing and are bounded
above by the corresponding partial sums from part (a), together with the
first term.  Apply the monotone convergence theorem.  For part (c), the
case $r=0$ is immediate.  If
$0<|r|<1$, put $c=1/|r|>1$ and prove by induction that
$c^N\geq1+N(c-1)$ for every $N\in\N$.  Use the Archimedean property to
deduce that $r^N\to0$.}

\subpart{B. --- Decimal representations}

A \emph{digit sequence} is a map
$d\colon\N\to\{0,1,\ldots,9\}$.  Its associated partial sums are
\[
  s_N=\sum_{n=1}^{N}\frac{d_n}{10^n}
  \qquad(N\in\N).
\]
If $(s_N)_{N\in\N}$ converges to $r\in\R$, we say that the digit sequence
\emph{represents} $r$, write $r=0.d_1d_2d_3\ldots$, and call this a
\emph{decimal expansion} of $r$.

\examquestion{\label{q:decimal-existence}
\begin{subquestions}
  \item \nameditem{DSC}{Digit Series Convergence}
  Let $(d_n)_{n\in\N}$ be a digit sequence, and let
  $(s_N)_{N\in\N}$ be its associated sequence of partial sums.  Prove that
  $(s_N)_{N\in\N}$ is increasing and bounded above by $1$.  Deduce that
  every digit sequence represents a real number in $[0,1]$.
  \item \nameditem{GDDC}{Greedy Decimal Digit Construction}
  Fix $x\in[0,1)$.  Starting with $t_0=0$, prove recursively that,
  for each $n\in\N$, there is a largest digit
  $e_n\in\{0,1,\ldots,9\}$ subject to
  \[
    t_{n-1}+e_n10^{-n}\leq x.
  \]
  Having chosen it, define $t_n=t_{n-1}+e_n10^{-n}$.
  \item \nameditem{GDEB}{Greedy Decimal Error Bound}
  Prove by induction that
  \[
    0\leq x-t_N<10^{-N}
    \qquad\hbox{for every $N\in\N_0$.}
  \]
  \item \nameditem{EDE}{Existence of Decimal Expansions}
  Deduce that $t_N\to x$.  Thus every $x\in[0,1)$ has a decimal
  expansion $x=0.e_1e_2e_3\ldots$.
\end{subquestions}}

\examquestion{\label{q:decimal-uniqueness}
Let $(a_j)_{j\in\N}$ and $(b_j)_{j\in\N}$ be digit sequences whose
decimal expansions first differ in position $k\in\N$.  After interchanging
the two sequences if necessary, assume that $a_k<b_k$.
\begin{subquestions}
  \item \nameditem{DTDB}{Decimal Tail Difference Bound}
  Prove
  \[
    \sum_{j=k+1}^{\infty}\frac{a_j-b_j}{10^j}
      \leq\frac1{10^k}.
  \]
  \item \nameditem{ECDTB}{Equality Case in the Decimal Tail Bound}
  If the two expansions have the same value, show that equality is
  forced throughout the estimate.
  \item \nameditem{TRNA}{Terminating and Repeating Nine Ambiguity}
  Deduce that $b_k=a_k+1$, every later $a_j$ equals $9$, and every
  later $b_j$ equals $0$.
  \item \nameditem{UDE}{Uniqueness of Decimal Expansions}
  State precisely the uniqueness theorem for decimal expansions.
\end{subquestions}
\indication{Compare the first unequal digit, whose contribution has
absolute size at least $10^{-k}$, with the largest possible tail.}}

A decimal expansion $0.d_1d_2d_3\ldots$ is \emph{eventually periodic} if
there are $\ell\in\N_0$ and $h\in\N$ such that $d_{n+h}=d_n$ for every
$n\in\N$ with $n>\ell$.  The positive integer $h$ is then called a
\emph{period length} after position $\ell$.

\examquestion{\label{q:periodic}
\begin{subquestions}
  \item \nameditem{EPDR}{Eventually Periodic Decimals are Rational}
  Sum a geometric series to prove that every eventually periodic
  decimal expansion represents a rational number.
  \item \nameditem{LDPDD}{Long Division Produces Decimal Digits}
  Let $a\in\N_0$ and $b\in\N$ satisfy $0\leq a<b$ and
  $\gcd(a,b)=1$.  Put $r_0=a$.  For each $n\in\N$, apply division with
  remainder to define integers $d_n,r_n\in\N_0$ by
  \[
    10r_{n-1}=bd_n+r_n,
    \qquad 0\leq r_n<b.
  \]
  Prove that $d_n\in\{0,1,\ldots,9\}$ and that, for every $N\in\N$,
  \[
    \frac ab=\sum_{n=1}^{N}\frac{d_n}{10^n}
      +\frac{r_N}{b10^N}.
  \]
  \item \nameditem{RDEP}{Rational Decimals are Eventually Periodic}
  Use the pigeonhole principle to show that the sequence of remainders
  either reaches $0$ or eventually repeats.  Deduce that $a/b$ has an
  eventually periodic decimal expansion.  Conclude that every member of
  $\Q\cap[0,1)$ has such an expansion.  Use the Division Algorithm on a
  numerator and its positive denominator to show that every non-negative
  rational has the form $m+x$, where $m\in\N_0$ and
  $x\in\Q\cap[0,1)$.  Thus its digits after the decimal point are eventually
  periodic; a negative rational is recorded by placing a minus sign before
  the corresponding positive representation.
  \item \nameditem{DPPB}{Decimal Preperiod and Period Bounds}
  Continue with the integers $a$ and $b$ from part (b).  Write
  $b=2^\alpha5^\beta q$, where
  $\alpha,\beta\in\N_0$, $q\in\N$, and $\gcd(q,10)=1$.  When $q>1$, use
  the Fermat--Euler theorem to prove that, after at most
  $\max(\alpha,\beta)$ non-periodic digits, there is a period length
  $h\in\N$ which divides $\phi(q)$.
\end{subquestions}
\indication{For part (d), let $t=\max(\alpha,\beta)$.  After multiplying
$a/b$ by $10^t$ and cancelling powers of $2$ and $5$, the remaining
denominator is $q$.  Apply the congruence
$10^{\phi(q)}\equiv1\pmod q$.}}

A real number is \emph{irrational} if it is not rational, that is, if it
does not belong to $\Q$.

\examquestion{\label{q:prime-digits}
For every $n\in\N$, define $d_n=1$ when $n$ is prime and $d_n=0$
otherwise, and let $x=0.d_1d_2d_3\ldots$ be the real number represented by
this digit sequence.
\begin{subquestions}
  \item \nameditem{PPEP}{Prime Position beyond an Eventual Preperiod}
  Suppose that the digits are eventually periodic after position
  $\ell\in\N_0$, with period length $h\in\N$.  Use Question
  \ref{q:euclid-primes} to choose a prime $p\in\N$ with $p>\ell$.
  \item \nameditem{PIDCP}{Prime-Indexed Digits Contradict Periodicity}
  Compare the digits in positions $p$ and $p+ph$ and obtain a
  contradiction.
  \item \nameditem{IPID}{Irrationality from Prime-Indexed Digits}
  Use Question \ref{q:periodic} to conclude that $x$ is irrational.
\end{subquestions}}

\subpart{C. --- Irrationality and transcendence}

\examquestion{\label{q:e-irrational}
For each $N\in\N$, put
\[
  e_N=\sum_{j=0}^{N}\frac1{j!}.
\]
\begin{subquestions}
  \item \nameditem{CBES}{Convergence and Bounds for the Exponential Series}
  Prove that $j!\geq2^{j-1}$ for every $j\in\N$ with $j\geq2$.
  Deduce that $(e_N)_{N\in\N}$ converges.  Denote its limit by
  \[
    e=\sum_{j=0}^{\infty}\frac1{j!},
  \]
  and prove that $2<e<3$.
  \item \nameditem{FMRA}{Factorial Multiple under a Rationality Assumption}
  Suppose $e=p/q$, where $p\in\Z$, $q\in\N$, and
  $\gcd(p,q)=1$; this is what \emph{lowest terms} means.
  Explain why $q\geq2$, and prove that $q!e$ is an integer.
  \item \nameditem{ITFS}{Integral Truncated Factorial Sum}
  Show that
  \[
    q!\sum_{j=0}^{q}\frac1{j!}
  \]
  is an integer.
  \item \nameditem{SBFT}{Strict Bound on the Factorial Tail}
  Prove
  \[
    0<q!\sum_{j=q+1}^{\infty}\frac1{j!}
      <\sum_{r=1}^{\infty}\frac1{(q+1)^r}
      =\frac1q<1.
  \]
  \item \nameditem{IEC}{Irrationality of the Exponential Constant}
  Obtain a contradiction and conclude that $e$ is irrational.
\end{subquestions}}

\concept{Complex numbers and polynomials}
Let $i$ be a formal symbol satisfying $i^2=-1$.  A \emph{complex number}
is a formal expression $a+bi$ with $a,b\in\R$.  If
$a,b,c,d\in\R$, define equality and the two operations by
\[
  a+bi=c+di\quad\Longleftrightarrow\quad a=c\text{ and }b=d,
\]
\[
  (a+bi)+(c+di)=(a+c)+(b+d)i,
\]
\[
  (a+bi)(c+di)=(ac-bd)+(ad+bc)i.
\]
Their set is
$\C=\{a+bi:a,b\in\R\}$.

A \emph{polynomial with integer coefficients} is a formal expression
\[
  P(X)=a_0+a_1X+\cdots+a_dX^d,
\]
where $d\in\N_0$ and $a_0,\ldots,a_d\in\Z$; the letter $X$ is called an
\emph{indeterminate}.  Two such expressions are equal when every
coefficient is equal, after trailing zero coefficients have been added if
necessary.  These polynomials form the set $\Z[X]$.  If
$a_d\ne0$, then $P$ is non-zero and has \emph{degree} $d$.  For
$z\in\C$, its value at $z$ is
$P(z)=a_0+a_1z+\cdots+a_dz^d$.  We identify each $r\in\R$ with the
complex number $r+0i\in\C$.  A complex
number $z\in\C$ is \emph{algebraic} if $P(z)=0$ for some non-zero
$P\in\Z[X]$.  A real number is \emph{transcendental} if it is not
algebraic.

\admitted{FTA}{Fundamental Theorem of Algebra}{If $d\in\N$,
$c_0,\ldots,c_d\in\C$, and $c_d\ne0$, then there exists $z\in\C$ such
that $c_0+c_1z+\cdots+c_dz^d=0$.}

\examquestion{\label{q:algebraic-estimates}
Let $d\in\N_0$ and let $a_0,a_1,\ldots,a_d\in\Z$, with $a_d\ne0$.
Put
\[
  P(X)=a_0+a_1X+\cdots+a_dX^d\in\Z[X].
\]
Thus $P$ is a non-zero polynomial of degree $d$.
\begin{subquestions}
  \item \nameditem{ERNA}{Every Rational Number is Algebraic}
  Let $r\in\Q$.  Write $r=p/q$ with $p\in\Z$ and $q\in\N$, and
  prove that $qX-p\in\Z[X]$ is a non-zero polynomial which vanishes at
  $r$.  Deduce that $r$ is algebraic.
  \item \nameditem{LBNPVR}{Lower Bound for Non-zero Polynomial Values at Rationals}
  If $r=p/q$ with $p\in\Z$, $q\in\N$, and $P(r)\ne0$, prove
  \[
    |P(r)|\geq q^{-d}.
  \]
  \item \nameditem{LBPVUI}{Lipschitz Bound for Polynomial Values on the Unit Interval}
  For $x,y\in[0,1]$ and each $j\in\N$ with $1\leq j\leq d$,
  factor $x^j-y^j$ and prove
  \[
    |P(x)-P(y)|\leq C_P|x-y|,
    \qquad C_P=\sum_{j=1}^d j|a_j|.
  \]
\end{subquestions}}

\admitted{PRB}{Polynomial Root Bound}{If $d\in\N_0$ and
$c_0,\ldots,c_d\in\C$ satisfy $c_d\ne0$, then the equation
$c_0+c_1z+\cdots+c_dz^d=0$ has at most $d$ solutions $z\in\C$.}

\examquestion{\label{q:liouville}
For each $m\in\N$, define
\[
  L_m=\sum_{n=1}^{m}10^{-n!}.
\]
\begin{subquestions}
  \item \nameditem{CBLS}{Convergence and Bounds for the Liouville Series}
  Prove that $(L_m)_{m\in\N}$ is increasing and bounded above.
  Denote its limit by
  \[
    L=\sum_{n=1}^{\infty}10^{-n!},
  \]
  and prove that $0<L<1$.
  \item \nameditem{QLTE}{Quantitative Liouville Tail Estimate}
  Prove that, for every $m\in\N$,
  \[
    0<L-L_m\leq\frac{10}{9}\,10^{-(m+1)!}.
  \]
  \item \nameditem{DBLPS}{Denominator Bound for Liouville Partial Sums}
  For every $m\in\N$, write $L_m=p_m/q_m$, where
  $p_m\in\Z$, $q_m\in\N$, and $\gcd(p_m,q_m)=1$.
  Prove that $q_m\leq10^{m!}$.
  \item \nameditem{NRA}{Non-vanishing at Rational Approximants}
  Let $d\in\N$, and suppose that a polynomial $P$ of degree $d$ as
  in Question \ref{q:algebraic-estimates} satisfies $P(L)=0$.  Explain why
  $P(L_m)\ne0$ for all sufficiently large $m\in\N$.
  \item \nameditem{CPVB}{Contradictory Polynomial Value Bounds}
  For every such $m\in\N$, combine the two estimates of Question
  \ref{q:algebraic-estimates} to prove
  \[
    10^{-dm!}
      \leq |P(L_m)|
      \leq \frac{10C_P}{9}\,10^{-(m+1)!}.
  \]
  \item \nameditem{TLN}{Transcendence of the Liouville Number}
  Choose $m\in\N$ with $m>d$ sufficiently large and obtain a
  contradiction.
  Conclude that $L$ is transcendental.
\end{subquestions}
\indication{For part (b), prove that
$n!\geq(m+1)!+(n-m-1)$ whenever $n\in\N$ and $n\geq m+1$, and compare
with a geometric series.  The truncations are strictly increasing, whereas
$P$ has only finitely many roots.  In part (f), use
$(m+1)!=(m+1)m!$ and compare the exponents after dividing the two sides.}}

\admitted{HLT}{Hermite Lindemann Transcendence}{The real number $e$ defined
in Question \ref{q:e-irrational} and the circle constant $\pi\in\R$ are
transcendental.}

\exampart{VIII}{Enumeration and diagonalisation}

This final part compares the sizes of infinite sets.  Recall that a set
$A$ is \emph{finite} if there exist $n\in\N_0$ and a bijection
$A\to[n]$.  The integer $n$, which is unique by Question
\ref{q:finite-cardinality}, is the \emph{cardinality} of $A$ and is denoted
by $|A|$.  A set is \emph{infinite} if it is not finite.

A set $A$ is \emph{countable} if it is finite or admits a bijection with
$\N$.  It is \emph{uncountable} if it is not countable.  An
\emph{enumeration} of a non-empty set $A$ is a surjective map
$e\colon\N\to A$.  If $e$ is bijective, the enumeration is a listing
without repetitions; writing $a_n=e(n)$ then gives the list
$a_1,a_2,a_3,\ldots$.
When a non-empty subset $S\subseteq\N$ has a least element $s$, we write
$s=\min S$.

\subpart{A. --- Countable constructions}

\examquestion{\label{q:countability-criteria}
\begin{subquestions}
  \item \nameditem{ESNN}{Enumeration of a Subset of the Natural Numbers}
  Let $S\subseteq\N$ be non-empty.  Set $s_1=\min S$.  If
  $k\in\N$ and $s_1,\ldots,s_k$ have been chosen while
  $S\setminus\{s_1,\ldots,s_k\}$ is non-empty, set
  \[
    s_{k+1}=\min\bigl(S\setminus\{s_1,\ldots,s_k\}\bigr).
  \]
  Prove that this process either stops and $S$ is finite, or continues and
  defines a bijection $\N\to S$ by $k\mapsto s_k$.
  \item \nameditem{ECC}{Equivalent Criteria for Countability}
  Prove that the following statements are equivalent for a set $A$:
  \begin{properties}
    \item $A$ is countable;
    \item there is an injection $A\to\N$;
    \item $A=\varnothing$ or there is a surjection $\N\to A$.
  \end{properties}
\end{subquestions}
\indication{In part (a), if some $t\in S$ were never selected, only
finitely many members of $S$ could be smaller than $t$; use this to obtain
a contradiction.  For (iii)$\Rightarrow$(ii), send each $a\in A$ to the least
element of its non-empty pre-image.  For (ii)$\Rightarrow$(i), apply part
(a) to the image of the injection.  Treat finite and infinite cases
explicitly.}}

For sets $A$ and $I$, write $A^I$ for the set of all maps $I\to A$.  If
$k\in\N_0$, write $A^k=A^{[k]}$; thus $A^k$ is the set of ordered
$k$-tuples of elements of $A$, and $A^0$ contains only the empty tuple.

\examquestion{\label{q:basic-countable}
\begin{subquestions}
  \item \nameditem{EBINN}{Explicit Bijection between the Integers and Natural Numbers}
  Construct an explicit bijection between $\Z$ and $\N$.
  \item \nameditem{CPB}{Cantor Pairing Bijection}
  For $(a,b)\in\N_0^2$, put
  \[
    \theta(a,b)=\frac{(a+b)(a+b+1)}2+b.
  \]
  Prove that $\theta$ is a bijection $\N_0^2\to\N_0$ by enumerating, for
  each $s\in\N_0$, the diagonal line formed by the pairs
  $(a,b)\in\N_0^2$ satisfying $a+b=s$.
  \item \nameditem{CINP}{Countability of Integer and Natural Pairs}
  Deduce that $\N^2$ and $\Z^2$ are countable.
  \item \nameditem{CFDIL}{Countability of Fixed-Dimensional Integer Lattices}
  Prove by induction that $\Z^k$ is countable for every fixed
  $k\in\N$.
\end{subquestions}}

The \emph{countable axiom of choice} says that, if
$(A_n)_{n\in\N}$ is a family of non-empty sets, then one may choose an
element $a_n\in A_n$ for every $n\in\N$ simultaneously.  This principle
is the special case of the axiom of choice in which the index set is
$\N$, and it is used implicitly in the corresponding theorem in the
original notes.

\examquestion{\label{q:countable-unions}
\begin{subquestions}
  \item \nameditem{SCSC}{Subsets of Countable Sets are Countable}
  Let $A$ and $B$ be sets.  If there is an injection $A\to B$ and
  $B$ is countable, prove that $A$ is countable.
  \item \nameditem{SNPCU}{Surjection from Natural Pairs onto a Countable Union}
  For every $n\in\N$, let $A_n$ be a non-empty set and let a
  surjection $s_n\colon\N\to A_n$ be specified.  Prove that the map
  \[
    F\colon\N^2\longrightarrow\bigcup_{n\in\N}A_n,
    \qquad F(n,m)=s_n(m),
  \]
  is surjective.
  \item \nameditem{CUSE}{Countable Union with Supplied Enumerations}
  For every $n\in\N$, let $A_n$ be a countable set, and suppose that
  an enumeration of every non-empty $A_n$ has been supplied.  Deduce that
  $\bigcup_{n\in\N}A_n$ is countable.
  \item \nameditem{CCGCUT}{Countable Choice Gives the Countable Union Theorem}
  Use the countable axiom of choice to explain how the enumerations
  required in part (c) may be chosen simultaneously.  State the resulting
  countable-union theorem.
\end{subquestions}
\indication{For part (c), the conclusion is immediate if the union is
empty.  Otherwise fix one element of the union and use it to replace each
empty member of the family by a singleton before applying part (b).}}

\examquestion{\label{q:finite-codes}
\begin{subquestions}
  \item \nameditem{CRN}{Countability of the Rational Numbers}
  Let $r\in\Q$.  Starting with an arbitrary representative of $r$,
  use its greatest common divisor to prove that there is a pair
  $(a,b)\in\Z\times\N$ such that
  \[
    r=a/b,\qquad \gcd(a,b)=1.
  \]
  Use unique prime factorisation, or Euclid's lemma, to prove that this
  pair is unique.  Send $r$ to $(a,b)$ and deduce that $\Q$ is countable.
  \item \nameditem{CFSNS}{Countability of Fixed-Size Natural Subsets}
  Let $k\in\N_0$.  Show that the set of $k$-element subsets of
  $\N$ is countable by listing the elements of each subset in increasing
  order.
  \item \nameditem{CFNS}{Countability of Finite Natural Subsets}
  Deduce that the set of all finite subsets of $\N$ is countable.
  \item \nameditem{CFWCA}{Countability of Finite Words over a Countable Alphabet}
  Let $\Sigma$ be a countable set.  A \emph{word of length $k$ over
  $\Sigma$}, where $k\in\N_0$, is a map $[k]\to\Sigma$; the unique word
  of length $0$ is the \emph{empty word}.  Put
  \[
    \Sigma^{<\N}=\bigcup_{k\in\N_0}\Sigma^k.
  \]
  Prove that the set $\Sigma^{<\N}$ of all finite words over $\Sigma$ is
  countable.
\end{subquestions}}

\examquestion{\label{q:algebraics-countable}
Let $\mathcal A$ be the set of algebraic real numbers.
\begin{subquestions}
  \item \nameditem{CBDIP}{Countability of Bounded-Degree Integer Polynomials}
  Let $d\in\N_0$.  Encode every polynomial
  $a_0+a_1X+\cdots+a_dX^d$, where $a_0,\ldots,a_d\in\Z$, by its
  $(d+1)$ coefficients.  Prove that this set, which includes the zero
  polynomial and every polynomial of degree at most $d$, is countable.
  \item \nameditem{IPRC}{Integer Polynomial Ring is Countable}
  Prove that $\Z[X]$ is countable.
  \item \nameditem{CARN}{Countability of Algebraic Real Numbers}
  Use the admitted fact that, for every $d\in\N_0$, a non-zero
  polynomial of degree $d$ has at most $d$ real roots, and prove that
  $\mathcal A$ is countable.
\end{subquestions}}

\subpart{B. --- Diagonal arguments}

\examquestion{\label{q:reals-uncountable}
The real numbers $1/(n+1)$, for $n\in\N$, are distinct elements of
$[0,1)$, so this interval is infinite.  Suppose, for a contradiction, that
$[0,1)$ is countable.  There is then a bijection $x\colon\N\to[0,1)$.
For each $i\in\N$, write $x_i=x(i)$ and choose the decimal expansion of
$x_i$ which does not end in an infinite string of $9$s; Questions
\ref{q:decimal-existence} and \ref{q:decimal-uniqueness} show that such a
choice exists and is unique.  For
$i,j\in\N$, denote its $j$th digit by $d_{ij}\in\{0,1,\ldots,9\}$.
\begin{subquestions}
  \item \nameditem{CDDC}{Cantor Decimal Diagonal Construction}
  For every $n\in\N$, define a digit $e_n$ by
  \[
    e_n=
    \begin{cases}
      1,&d_{nn}\ne1,\\
      2,&d_{nn}=1.
    \end{cases}
  \]
  Let $y=0.e_1e_2e_3\ldots$ be the real number represented by this digit
  sequence.
  \item \nameditem{DDAELR}{Diagonal Decimal Avoids Every Listed Real}
  Prove that $y\in[0,1)$ and $y\ne x_n$ for every $n\in\N$.
  Explain why the exceptional double decimal representations cause no
  problem.
  \item \nameditem{URN}{Uncountability of the Real Numbers}
  Conclude that $[0,1)$ is uncountable.  Use the inclusion
  $[0,1)\subseteq\R$ and Question \ref{q:countable-unions}(a) to deduce
  that $\R$ is uncountable.
\end{subquestions}}

\examquestion{\label{q:many-transcendentals}
\begin{subquestions}
  \item \nameditem{FUCS}{Finite Union of Countable Sets}
  Let $A$ and $B$ be countable sets.  Prove directly from the
  countability criteria that $A\cup B$ is countable.
  \item \nameditem{UMTRN}{Uncountably Many Transcendental Real Numbers}
  Since $\R=\mathcal A\cup(\R\setminus\mathcal A)$, use Questions
  \ref{q:algebraics-countable} and \ref{q:reals-uncountable} to prove that
  the set of transcendental real numbers is uncountable.
  \item \nameditem{AECT}{Abundance versus Explicit Construction of Transcendentals}
  Explain why this argument proves that there are uncountably many
  transcendental numbers, while the explicit construction in Question
  \ref{q:liouville} identifies only one of them.
\end{subquestions}}

For a subset $S\subseteq\N$, its \emph{indicator} is the map
$\mathbf1_S\colon\N\to\{0,1\}$ defined, for every $n\in\N$, by
\[
  \mathbf1_S(n)=
  \begin{cases}
    1,&n\in S,\\
    0,&n\notin S.
  \end{cases}
\]
The set $\{0,1\}^{\N}$ is therefore the set of all binary sequences, and
$\N^{\N}$ is the set of all maps $\N\to\N$.

\examquestion{\label{q:sequence-diagonals}
\begin{subquestions}
  \item \nameditem{SNNBS}{Subsets of Natural Numbers are Binary Sequences}
  Prove that the map
  \[
    S\longmapsto\mathbf1_S
  \]
  is a bijection from $\mathcal P(\N)$ to the set $\{0,1\}^{\N}$ of binary
  sequences.
  \item \nameditem{DSAELS}{Diagonal Subset Avoids Every Listed Subset}
  For every $n\in\N$, let $S_n$ be a subset of $\N$.  Define
  \[
    S=\{n\in\N:n\notin S_n\}
  \]
  and prove that $S\ne S_n$ for every $n\in\N$.
  \item \nameditem{DFAELF}{Diagonal Function Avoids Every Listed Function}
  For every $n\in\N$, let $f_n\colon\N\to\N$ be a map.  Define
  $f\colon\N\to\N$ by $f(n)=f_n(n)+1$ for every $n\in\N$.  Prove that
  $f\ne f_n$ for every $n\in\N$.
  \item \nameditem{UPSFS}{Uncountability of Power Sets and Function Spaces}
  Conclude that both $\mathcal P(\N)$ and $\N^{\N}$ are uncountable.
\end{subquestions}}

Two sets $A$ and $B$ have the \emph{same cardinality} if there is a
bijection $A\to B$.  We say that $B$ has cardinality \emph{at least as
large as} that of $A$ if there is an injection $A\to B$, and that $B$ has
\emph{strictly larger cardinality} than $A$ if such an injection exists but
no bijection $A\to B$ exists.

\examquestion{\label{q:cantor}
Let $A$ be any set and suppose $f\colon A\to\mathcal P(A)$ is surjective.
\begin{subquestions}
  \item \nameditem{CDC}{Cantor Diagonal Contradiction}
  Define
  \[
    D=\{a\in A:a\notin f(a)\}.
  \]
  Choose $d\in A$ with $f(d)=D$ and show that
  $d\in D\Longleftrightarrow d\notin D$.
  \item \nameditem{NSPS}{No Surjection onto a Power Set}
  Conclude that there is no surjection $A\to\mathcal P(A)$.
  \item \nameditem{CPST}{Cantor Power Set Theorem}
  Prove that the map
  $i\colon A\to\mathcal P(A)$ defined by $i(a)=\{a\}$ for every
  $a\in A$ is injective.  Deduce that
  $\mathcal P(A)$ has strictly larger cardinality than $A$.
\end{subquestions}}

\examquestion{\label{q:csb}
Let $A$ and $B$ be sets, and let $f\colon A\to B$ and $g\colon B\to A$
be injections.  Define
\[
  A_0=A\setminus g(B),\qquad
  A_{n+1}=g(f(A_n))\quad(n\in\N_0),\qquad
  A_*=\bigcup_{n\in\N_0}A_n.
  \tag{$\star$}
\]
\begin{subquestions}
  \item \nameditem{UPIR}{Unique Pre-images outside the Iterated Region}
  Prove that every $a\in A\setminus A_*$ belongs to $g(B)$, and hence
  has a unique pre-image in $B$.  For each such $a$, denote this unique
  element by $\gamma(a)\in B$, so that $g(\gamma(a))=a$.  Prove that this
  defines a function $\gamma\colon A\setminus A_*\to B$.
  \item \nameditem{PCSBMWD}{Piecewise Cantor Schroder Bernstein Map is Well-Defined}
  Define
  \[
    h\colon A\to B,\qquad
    h(a)=
    \begin{cases}
      f(a),&a\in A_*,\\
      \gamma(a),&a\notin A_*.
    \end{cases}
  \]
  Prove that $h$ is well-defined.
  \item \nameditem{ICSBM}{Injectivity of the Cantor Schroder Bernstein Map}
  Prove that $h$ is injective.  In the mixed case, use
  $g(f(A_n))=A_{n+1}$ for $n\in\N_0$.
  \item \nameditem{SCSBM}{Surjectivity of the Cantor Schroder Bernstein Map}
  Let $b\in B$.  Treat separately the cases $g(b)\notin A_*$ and
  $g(b)\in A_*$ to prove that $b$ lies in the image of $h$.
  \item \nameditem{CSBT}{Cantor Schroder Bernstein Theorem}
  Conclude the Cantor--Schr\"oder--Bernstein theorem: injections in
  both directions imply the existence of a bijection.
\end{subquestions}
\indication{If $g(b)\in A_*$, it cannot belong to $A_0$; use $(\star)$ and
the injectivity of $g$ to write $b=f(a)$ for some $a\in A_*$.}}

\examquestion{\label{q:continuum-power-set}
Question \ref{q:finite-codes}(a) shows that $\Q$ is countable, and the
inclusion $\N\subseteq\Q$ shows that it is infinite.  Fix a bijection
$r\colon\N\to\Q$, and write $r_n=r(n)$ for every $n\in\N$.
\begin{subquestions}
  \item \nameditem{IRNPS}{Injection from the Reals into the Natural Power Set}
  Define
  \[
    \Phi\colon\R\to\mathcal P(\N),\qquad
    \Phi(x)=\{n\in\N:r_n<x\}.
  \]
  Use the density of $\Q$ in $\R$ to prove that $\Phi$ is injective.
  \item \nameditem{TENSR}{Ternary Encoding of Natural Subsets as Reals}
  For every $S\subseteq\N$ and $N\in\N$, define
  \[
    T_N(S)=\sum_{n=1}^{N}\frac{2\mathbf1_S(n)}{3^n}.
  \]
  Prove that $(T_N(S))_{N\in\N}$ converges.  Define
  \[
    \Psi\colon\mathcal P(\N)\to\R,
    \qquad \Psi(S)=\lim_{N\to\infty}T_N(S).
  \]
  \item \nameditem{ITSE}{Injectivity of the Ternary Subset Encoding}
  Let $S,T\subseteq\N$ with $S\ne T$, and let $k\in\N$ be the least
  element of
  $S\triangle T$.  Compare the $k$th contribution with the entire remaining
  tail and prove $\Psi(S)\ne\Psi(T)$.
  \item \nameditem{ECNPS}{Equinumerosity of the Continuum and the Natural Power Set}
  Apply Question \ref{q:csb} to prove that $\R$ and
  $\mathcal P(\N)$ have the same cardinality.
\end{subquestions}
\indication{For part (c),
$2/3^k>\sum_{n=k+1}^{\infty}2/3^n=1/3^k$.}}

\examquestion{\label{q:cardinal-hierarchy}
\begin{subquestions}
  \item \nameditem{IPSGICH}{Iterated Power Sets Give an Infinite Cardinal Hierarchy}
  Let $A$ be a set.  Apply Question \ref{q:cantor} repeatedly
  to the chain of sets
  \[
    A,\quad\mathcal P(A),\quad
    \mathcal P(\mathcal P(A)),\quad\ldots
  \]
  and prove that every term has strictly larger cardinality than its
  predecessor.
  \item \nameditem{NUC}{Non-existence of a Universal Cardinal}
  Deduce that there is no set $U$ whose cardinality is at least as
  large as that of every set.
  \item \nameditem{CHCGS}{Continuum Hypothesis as a Cardinal Gap Statement}
  The \emph{continuum hypothesis} asserts that there is no set $B$
  such that $B$ has strictly larger cardinality than $\N$ and
  $\mathcal P(\N)$ has strictly larger cardinality than $B$.  In view of
  Question \ref{q:continuum-power-set}, explain why this is equivalent to
  saying that no cardinality lies strictly between those of $\N$ and
  $\R$.
\end{subquestions}}
\indication{For part (b), apply the asserted universal property to
$\mathcal P(U)$.  Combine the resulting injection
$\mathcal P(U)\to U$ with the singleton injection $U\to\mathcal P(U)$,
and use Question \ref{q:csb}.}

\admitted{CHI}{Continuum Hypothesis Independence}{Zermelo--Fraenkel set theory
with the axiom of choice is commonly abbreviated to ZFC.  Assuming that ZFC
is consistent, neither the continuum hypothesis nor its negation can be
proved from the axioms of ZFC.\hfill\emph{End of the problem}}
